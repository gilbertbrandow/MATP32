\section*{Problem 1}

\subsection*{(a)}

\begin{proof}
\begin{align*}
  M'_c(f)(x)
  &= \sup_{r>0}\frac{1}{r^d}\int_{x+(-r/2,r/2)^d}|f(y)|\,dy \\
  &\leq \sup_{r>0}\frac{1}{r^d}\int_{B(x,\,r\sqrt{d}/2)}|f(y)|\,dy
    &&\text{($x+(-r/2,r/2)^d \subseteq B(x,r\sqrt{d}/2)$)} \\
  &= \sup_{\rho>0}\frac{d^{d/2}}{2^d\rho^d}\int_{B(x,\rho)}|f(y)|\,dy
    &&\text{($\rho = r\sqrt{d}/2$)} \\
  &= d^{d/2}\cdot v_d\cdot 2^{-d} \sup_{\rho>0}\frac{1}{v_d\rho^d}\int_{B(x,\rho)}|f(y)|\,dy \\
  &= d^{d/2}\cdot v_d\cdot 2^{-d}\cdot M_c(f)(x). \qedhere
\end{align*}
\end{proof}

\subsection*{(b)}

\begin{proof}
\begin{align*}
  M_c(f)(x)
  &= \sup_{r>0}\frac{1}{v_d r^d}\int_{B(x,r)}|f(y)|\,dy \\
  &\leq \sup_{r>0}\frac{1}{v_d r^d}\int_{x+(-r,r)^d}|f(y)|\,dy
    &&\text{($B(x,r) \subseteq x+(-r,r)^d$)} \\
  &= \sup_{s>0}\frac{2^d}{v_d s^d}\int_{x+(-s/2,s/2)^d}|f(y)|\,dy
    &&\text{($s = 2r$)} \\
  &= 2^d\cdot v_d^{-1} \sup_{s>0}\frac{1}{s^d}\int_{x+(-s/2,s/2)^d}|f(y)|\,dy \\
  &= 2^d\cdot v_d^{-1}\cdot M'_c(f)(x). \qedhere
\end{align*}
\end{proof}

\section*{Problem 2}

\subsection*{(a)}

\begin{proof}
Since clearly $|f(x)-q| \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$, by Lebesgue Differentiation Theorem
\begin{equation}\label{eq:ldt-q}
  \lim_{r\to 0^+}\frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-q|\,dy = |f(x)-q| \quad \text{for a.e. } x \in \mathbb{R}^d.
\end{equation}
Let $A_q$ be the set where it fails, then $|A_q| = 0$ and the limit holds for all $x \in \mathbb{R}^d \setminus A_q$.
\end{proof}

\subsection*{(b)}

\begin{proof}
Let $A := \bigcup_{q\in\mathbb{Q}} A_q$, where each $A_q$ is as in previous part.
Since $\mathbb{Q}$ is countable, $|A| = 0$, and for every $x \in \mathbb{R}^d \setminus A$,
\eqref{eq:ldt-q} holds simultaneously for all $q \in \mathbb{Q}$.

Fix $x \in \mathbb{R}^d \setminus A$ and $\varepsilon > 0$. Since $\mathbb{Q}$ is dense in $\mathbb{R}$, choose
$q \in \mathbb{Q}$ such that $|f(x)-q| < \varepsilon/2$. Because of \eqref{eq:ldt-q}, there exists $r_0 > 0$
such that for all $r \in (0, r_0)$,
\[
  \frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-q|\,dy < \frac{\varepsilon}{2}.
\]
Then for all $r \in (0, r_0)$,
\begin{align*}
  \frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-f(x)|\,dy
  &\leq \frac{1}{|B(x,r)|}\int_{B(x,r)}(|f(y)-q| + |f(x)-q|)dy \\
  &= \frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-q|\,dy + |f(x)-q| \\
  &< \frac{\varepsilon}{2} + \frac{\varepsilon}{2} = \varepsilon.
\end{align*}
Since $\varepsilon > 0$ was arbitrary,
\begin{equation}\label{eq:ldt-b}
  \lim_{r\to 0^+}\frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-f(x)|\,dy = 0 \quad \forall x \in \mathbb{R}^d \setminus A. \qedhere
\end{equation}
\end{proof}

\subsection*{(c)}

\begin{proof}
Let $A$ be the set from previous part and let $r_n := \sqrt{d}\cdot 2^{-n}$, so $Q_n(x) \subseteq B(x,r_n)$,
with $|Q_n| = 2^{-nd}$ and $|B(x,r_n)| = v_d r_n^d = v_d d^{d/2} \cdot 2^{-nd}$.

Fix $x \in \mathbb{R}^d \setminus A$ and $\varepsilon > 0$; by \eqref{eq:ldt-b}, $\exists N \in \mathbb{N}$ such that $\forall n \geq N$,
\[
  \frac{1}{|B(x,r_n)|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy < \frac{\varepsilon}{v_d d^{d/2}}.
\]
Then for all $n \geq N$,
\begin{align*}
  \left|\frac{1}{|Q_n|}\int_{Q_n} f(y)\,dy - f(x)\right|
  &\leq \frac{1}{|Q_n|}\int_{Q_n}|f(y)-f(x)|\,dy \\
  &\leq \frac{1}{|Q_n|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy \\
  &= \frac{|B(x,r_n)|}{|Q_n|}\cdot\frac{1}{|B(x,r_n)|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy \\
  &= \frac{v_d r_n^d}{2^{-nd}}\cdot\frac{1}{|B(x,r_n)|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy \\
  &= \frac{v_d \cdot d^{d/2} \cdot 2^{-nd}}{2^{-nd}}\cdot\frac{1}{|B(x,r_n)|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy \\
  &= v_d d^{d/2}\cdot\frac{1}{|B(x,r_n)|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy \\
  &< v_d d^{d/2}\cdot \frac{\varepsilon}{v_d d^{d/2}} = \varepsilon.
\end{align*}
Since $\varepsilon > 0$ was arbitrary, $\left|\frac{1}{|Q_n|}\int_{Q_n} f(y)\,dy - f(x)\right| \to 0$ as $n\to\infty$,
which is equivalent to $\frac{1}{|Q_n(x)|}\int_{Q_n(x)} f(y)\,dy \to f(x)$, $\forall x \in \mathbb{R}^d \setminus A$.
\end{proof}

\section*{Problem 3}

\subsection*{(a)}

\begin{proof}
For $\delta>0$, let $f_\delta(x_1,x_2):=\delta^{-2}\chi_{(-\delta,\delta)}(x_1)\chi_{(-\delta,\delta)}(x_2)$. Then $\|f_\delta\|_{L^1} = \delta^{-2}\cdot|(-\delta,\delta)^2| = \delta^{-2}\cdot(2\delta)^2 = 4.$
For $x=(x_1,x_2)$ with $x_1,x_2>\delta$, define the rectangle $R_x^\delta := [-\delta,\,x_1]\times[-\delta,\,x_2] \in \mathcal{R},$
with $|R_x^\delta| = (x_1+\delta)(x_2+\delta)$.

Suppose for contradiction there exists $C>0$ such that $|\{M_S(f)>\alpha\}|\leq\frac{C}{\alpha}\|f\|_{L^1}$
for all $f\in L^1(\mathbb{R}^2)$ and $\alpha>0$. Fix $\alpha>0$ and choose $\delta_0 < e^{-(C+4)/8}\alpha^{-1/2}$. Then:
\begin{align*}
  |\{M_S(f_{\delta_0})>\alpha\}|
  &= 4\,|\{x_1,x_2>0 : M_S(f_{\delta_0})(x_1,x_2)>\alpha\}|
  &&\text{(symmetry, 4 quadrants)} \\
  &\geq 4\,|\{x_1,x_2>\delta_0 : M_S(f_{\delta_0})(x_1,x_2)>\alpha\}| \\
  &\geq 4\,|\{x_1,x_2>\delta_0 : \frac{1}{|R_x^{\delta_0}|}\int_{R_x^{\delta_0}} f_{\delta_0}>\alpha\}| \\
  &= 4\,\Bigl|\Bigl\{x_1,x_2>\delta_0 : \frac{4}{(x_1+\delta_0)(x_2+\delta_0)}>\alpha\Bigr\}\Bigr| \\
    &= 4\,|\{u,v>2\delta_0:uv<\tfrac{4}{\alpha}\}|
    &&\text{($u=x_1+\delta_0$, $v=x_2+\delta_0$)} \\
  &= 4\int_{2\delta_0}^{\infty} \Bigl|\bigl\{v>2\delta_0 : v<\tfrac{4}{\alpha u}\bigr\}\Bigr|\,du\\
  &= 4\int_{2\delta_0}^{2/(\alpha\delta_0)}\left(\frac{4}{\alpha u}-2\delta_0\right)du\\
  &= \frac{16}{\alpha}\ln\frac{1}{\alpha\delta_0^2} - \frac{16}{\alpha} + 16\delta_0^2
  > \frac{4C}{\alpha},
\end{align*}
contradicting $|\{M_S(f_{\delta_0})>\alpha\}|\leq\frac{C}{\alpha}\|f_{\delta_0}\|_{L^1}=\frac{4C}{\alpha}$. Hence $M_S$ is not of weak-type $(1,1)$.
\end{proof}

\subsection*{(b)}

\begin{proof}
For $g\in L^1_{\mathrm{loc}}(\mathbb{R})$, let
\[
  M(g)(x) := \sup_{\substack{I\ni x \\ I \text{ open interval}}} \frac{1}{|I|}\int_I |g(t)|\,dt.
\]
Since $M$ is weak-type $(1,1)$ and bounded on $L^\infty(\mathbb{R})$, Marcinkiewicz gives
$\|M(g)\|_{L^p(\mathbb{R})}\leq C_p\|g\|_{L^p(\mathbb{R})}$ for $1<p<\infty$; at $p=\infty$ this holds trivially, with constant $1$, directly from the definition. So
$\|M(g)\|_{L^p(\mathbb{R})}\leq C_p\|g\|_{L^p(\mathbb{R})}$ for all $1<p\leq\infty$.

For $t\in\mathbb{R}$, write $f_t(s):=f(t,s)$. By Fubini, $M$ applied in either coordinate direction is bounded on $L^p(\mathbb{R}^2)$ with the same constant $C_p$, for $1<p\le\infty$.
\begin{align*}
  \|M_S(f)\|_{L^p}
  &= \left\|\sup_{I_1\times I_2\ni x}\frac{1}{|I_1\times I_2|}\int_{I_1\times I_2}|f|\right\|_{L^p} \\
  &= \left\|\sup_{I_1\times I_2\ni x}\frac{1}{|I_1|}\int_{I_1}\!\frac{1}{|I_2|}\int_{I_2}\!|f_t(s)|\,ds\,dt\right\|_{L^p} \\
  &\leq \left\|\sup_{I_1\ni x_1}\frac{1}{|I_1|}\int_{I_1}M(f_t)(x_2)\,dt\right\|_{L^p} \\
  &= \left\|M(M(f_t)(x_2))(x_1)\right\|_{L^p} \\
  &\leq C_p\|M(f_t)(x_2)\|_{L^p} \\
  &\leq C_p^2\|f\|_{L^p}. \qedhere
\end{align*}
\end{proof}
