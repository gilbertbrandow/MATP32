\subsection*{(a)}

\begin{align*}
  M'_c(f)(x)
  &= \sup_{r>0}\frac{1}{r^d}\int_{x+(-r/2,r/2)^d}|f(y)|\,dy \\
  &\leq \sup_{r>0}\frac{1}{r^d}\int_{B(x,\,r\sqrt{d}/2)}|f(y)|\,dy
    &&\text{($x+(-r/2,r/2)^d \subseteq B(x,r\sqrt{d}/2)$)} \\
  &= \sup_{\rho>0}\frac{d^{d/2}}{2^d\rho^d}\int_{B(x,\rho)}|f(y)|\,dy
    &&\text{($\rho = r\sqrt{d}/2$)} \\
  &= d^{d/2}\cdot v_d\cdot 2^{-d} \sup_{\rho>0}\frac{1}{v_d\rho^d}\int_{B(x,\rho)}|f(y)|\,dy \\
  &= d^{d/2}\cdot v_d\cdot 2^{-d}\cdot M_c(f)(x).
\end{align*}

  &= 2^d\cdot v_d^{-1}\cdot M'_c(f)(x).
\end{align*}

\section*{Problem 2}

\subsection*{(a)}

Since clearly $g_q(x) := |f(x)-q| \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$, the Lebesgue Differentiation Theorem gives
\begin{equation}\label{eq:ldt-q}
  \lim_{r\to 0^+}\frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-q|\,dy = |f(x)-q| \quad \text{for a.e. } x \in \mathbb{R}^d.
\end{equation}
Let $A_q$ be the null set where this fails. Then $|A_q| = 0$ and the limit holds for all $x \in \mathbb{R}^d \setminus A_q$.

\subsection*{(b)}

Let $A := \bigcup_{q\in\mathbb{Q}} A_q$, where each $A_q$ is as in previous part.
Since $\mathbb{Q}$ is countable, $|A| = 0$, and for every $x \in \mathbb{R}^d \setminus A$,
\eqref{eq:ldt-q} holds simultaneously for all $q \in \mathbb{Q}$. Fix $x \in \mathbb{R}^d \setminus A$ and $\varepsilon > 0$.
Since $f$ is real-valued and $\mathbb{Q}$ is dense in $\mathbb{R}$, choose $q \in \mathbb{Q}$ such that $|f(x)-q| < \varepsilon/2$.
\begin{align*}
  &\limsup_{r\to 0^+}\frac{1}{|B(x,r)|}\int_{B(x,r)}\!|f(y)-f(x)|\,dy \\
  &\leq \limsup_{r\to 0^+}\frac{1}{|B(x,r)|}\int_{B(x,r)}\!\bigl(|f(y)-q|+|f(x)-q|\bigr)\,dy
    &&\\
  &= \limsup_{r\to 0^+}\frac{1}{|B(x,r)|}\int_{B(x,r)}\!|f(y)-q|\,dy + |f(x)-q|
    &&\text{($|f(x)-q|$ constant\ in $y$)}\\
  &= |f(x)-q| + |f(x)-q| = 2|f(x)-q| < \varepsilon.
\end{align*}
Since $\varepsilon > 0$ was arbitrary we are done.
