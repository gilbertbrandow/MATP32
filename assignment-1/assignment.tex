\section*{Problem 1}

\subsection*{(a)}

\begin{proof}
\begin{align*}
  M'_c(f)(x)
  &= \sup_{r>0}\frac{1}{r^d}\int_{x+(-r/2,r/2)^d}|f(y)|\,dy \\
  &\leq \sup_{r>0}\frac{1}{r^d}\int_{B(x,\,r\sqrt{d}/2)}|f(y)|\,dy
    &&\text{($x+(-r/2,r/2)^d \subseteq B(x,r\sqrt{d}/2)$)} \\
  &= \sup_{\rho>0}\frac{d^{d/2}}{2^d\rho^d}\int_{B(x,\rho)}|f(y)|\,dy
    &&\text{($\rho = r\sqrt{d}/2$)} \\
  &= d^{d/2}\cdot v_d\cdot 2^{-d} \sup_{\rho>0}\frac{1}{v_d\rho^d}\int_{B(x,\rho)}|f(y)|\,dy \\
  &= d^{d/2}\cdot v_d\cdot 2^{-d}\cdot M_c(f)(x). \qedhere
\end{align*}
\end{proof}

\subsection*{(b)}

\begin{proof}
\begin{align*}
  M_c(f)(x)
  &= \sup_{r>0}\frac{1}{v_d r^d}\int_{B(x,r)}|f(y)|\,dy \\
  &\leq \sup_{r>0}\frac{1}{v_d r^d}\int_{x+(-r,r)^d}|f(y)|\,dy
    &&\text{($B(x,r) \subseteq x+(-r,r)^d$)} \\
  &= \sup_{s>0}\frac{2^d}{v_d s^d}\int_{x+(-s/2,s/2)^d}|f(y)|\,dy
    &&\text{($s = 2r$)} \\
  &= 2^d\cdot v_d^{-1} \sup_{s>0}\frac{1}{s^d}\int_{x+(-s/2,s/2)^d}|f(y)|\,dy \\
  &= 2^d\cdot v_d^{-1}\cdot M'_c(f)(x). \qedhere
\end{align*}
\end{proof}

\section*{Problem 2}

\subsection*{(a)}

\begin{proof}
Since clearly $g_q(x) := |f(x)-q| \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$, by Lebesgue Differentiation Theorem
\begin{equation}\label{eq:ldt-q}
  \lim_{r\to 0^+}\frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-q|\,dy = |f(x)-q| \quad \text{for a.e. } x \in \mathbb{R}^d.
\end{equation}
Let $A_q$ be the set where it fails, then $|A_q| = 0$ and the limit holds for all $x \in \mathbb{R}^d \setminus A_q$.
\end{proof}

\subsection*{(b)}

\begin{proof}
Let $A := \bigcup_{q\in\mathbb{Q}} A_q$, where each $A_q$ is as in previous part.
Since $\mathbb{Q}$ is countable, $|A| = 0$, and for every $x \in \mathbb{R}^d \setminus A$,
\eqref{eq:ldt-q} holds simultaneously for all $q \in \mathbb{Q}$.

Fix $x \in \mathbb{R}^d \setminus A$ and $\varepsilon > 0$. Since $\mathbb{Q}$ is dense in $\mathbb{R}$, choose
$q \in \mathbb{Q}$ such that $|f(x)-q| < \varepsilon/3$. Because of \eqref{eq:ldt-q}, there exists $r_0 > 0$
such that for all $0 < r < r_0$,
\[
  \frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-q|\,dy < \frac{2\varepsilon}{3}.
\]
Then for all $0 < r < r_0$,
\begin{align*}
  \frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-f(x)|\,dy
  &\leq \frac{1}{|B(x,r)|}\int_{B(x,r)}(|f(y)-q| + |f(x)-q|)dy \\
  &= \frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-q|\,dy + |f(x)-q| \\
  &< \frac{2\varepsilon}{3} + \frac{\varepsilon}{3} = \varepsilon.
\end{align*}
Since $\varepsilon > 0$ was arbitrary,
\begin{equation}\label{eq:ldt-b}
  \lim_{r\to 0^+}\frac{1}{|B(x,r)|}\int_{B(x,r)}|f(y)-f(x)|\,dy = 0 \quad \forall x \in \mathbb{R}^d \setminus A. \qedhere
\end{equation}
\end{proof}

\subsection*{(c)}

\begin{proof}
Let $A$ be the set from previous part and let $r_n := \sqrt{d}\cdot 2^{-n}$, so $Q_n(x) \subseteq B(x,r_n)$,
with $|Q_n| = 2^{-nd}$ and $|B(x,r_n)| = v_d r_n^d = v_d d^{d/2} \cdot 2^{-nd}$.

Fix $x \in \mathbb{R}^d \setminus A$ and $\varepsilon > 0$; by \eqref{eq:ldt-b}, $\exists N \in \mathbb{N}$ such that $\forall n \geq N$,
\[
  \frac{1}{|B(x,r_n)|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy < \frac{\varepsilon}{v_d d^{d/2}}.
\]
Then for all $n \geq N$,
\begin{align*}
  \left|\frac{1}{|Q_n|}\int_{Q_n} f(y)\,dy - f(x)\right|
  &\leq \frac{1}{|Q_n|}\int_{Q_n}|f(y)-f(x)|\,dy \\
  &\leq \frac{1}{|Q_n|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy \\
  &= \frac{|B(x,r_n)|}{|Q_n|}\cdot\frac{1}{|B(x,r_n)|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy \\
  &= \frac{v_d r_n^d}{2^{-nd}}\cdot\frac{1}{|B(x,r_n)|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy \\
  &= \frac{v_d \cdot d^{d/2} \cdot 2^{-nd}}{2^{-nd}}\cdot\frac{1}{|B(x,r_n)|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy \\
  &= v_d d^{d/2}\cdot\frac{1}{|B(x,r_n)|}\int_{B(x,r_n)}|f(y)-f(x)|\,dy \\
  &< v_d d^{d/2}\cdot \frac{\varepsilon}{v_d d^{d/2}} = \varepsilon.
\end{align*}
Since $\varepsilon > 0$ was arbitrary, $\left|\frac{1}{|Q_n|}\int_{Q_n} f(y)\,dy - f(x)\right| \to 0$ as $n\to\infty$,
which is equivalent to $\frac{1}{|Q_n(x)|}\int_{Q_n(x)} f(y)\,dy \to f(x)$, $\forall x \in \mathbb{R}^d \setminus A$.
\end{proof}
