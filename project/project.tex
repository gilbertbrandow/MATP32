\section{Introduction}

\begin{definition}[Weights]
A \emph{weight} is a function $w \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$ with $w \geq 0$. For a measurable set $E \subseteq \mathbb{R}^d$ we write $w(E) := \int_E w(x)\,dx$.
\end{definition}

\begin{definition}[Muckenhoupt $A_p$ weights]
Let $1 < p < \infty$ and let $w$ be a weight. The \emph{$A_p$ characteristic} of $w$ is
\begin{equation}\label{eq:Ap}
  [w]_{A_p} := \sup_{Q} \left( \frac{1}{|Q|} \int_Q w(x)\,dx \right) \left( \frac{1}{|Q|} \int_Q w(x)^{-\frac{1}{p-1}}\,dx \right)^{p-1},
\end{equation}
where the supremum is taken over all cubes $Q \subset \mathbb{R}^d$ with sides parallel to the axes. We say that $w \in A_p$ if $[w]_{A_p} < \infty$.
\end{definition}

\begin{definition}[Hardy-Littlewood maximal operator with respect to $w$]
Let $w$ be a weight. For a measurable function $f$ and $x \in \mathbb{R}^d$,
\begin{equation}\label{eq:M}
  M_w f(x) := \sup_{Q \ni x} \frac{1}{w(Q)} \int_Q |f(y)|\, w(y)\,dy,
\end{equation}
where the supremum is taken over all cubes $Q$ containing $x$. If the supremum is taken only over cubes $Q$ centred at $x$, we denote the corresponding operator by $M^c_w$. When $w \equiv 1$ we drop the subscript and write $M$ and $M^c$.
\end{definition}
