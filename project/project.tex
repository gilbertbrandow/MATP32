\section{Introduction}

\begin{definition}[Weights]
A \emph{weight} is a function $w \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$ with $w \geq 0$. For a measurable set $E \subseteq \mathbb{R}^d$ we write $w(E) := \int_E w(x)\,dx$.
\end{definition}

\begin{definition}[Weighted $L^p$ spaces]
Let $w$ be a weight and let $1 \le p < \infty$. We define
\[
  L^p_w(\mathbb{R}^d) := \left\{ f : \mathbb{R}^d \to \mathbb{R} \text{ measurable} : \int_{\mathbb{R}^d} |f(x)|^p\, w(x)\,dx < \infty \right\},
\]
with norm $\|f\|_{L^p_w} := \left( \int_{\mathbb{R}^d} |f(x)|^p\, w(x)\,dx \right)^{1/p}$. Moreover, $L^\infty_w(\mathbb{R}^d)$ is the space of measurable functions $f$ with $\|f\|_{L^\infty_w} := \inf\{ t \ge 0 : w(\{|f| > t\}) = 0 \} < \infty$.
\end{definition}

The central object of this report is the Hardy--Littlewood maximal operator $M$, which assigns to each point the largest average of $|f|$ over cubes containing that point. In the proof of our main result we will also need the analogous operator in which these averages are taken with respect to the measure $w(x)\,dx$ instead of Lebesgue measure.

\begin{definition}[Hardy--Littlewood maximal operator]
Let $w$ be a weight with $w(Q) > 0$ for every cube $Q$. For a measurable function $f$ and $x \in \mathbb{R}^d$,
\[
  M_w f(x) := \sup_{Q \ni x} \frac{1}{w(Q)} \int_Q |f(y)|\, w(y)\,dy,
\]
where the supremum is taken over all cubes $Q \subset \mathbb{R}^d$ with sides parallel to the axes containing $x$. If the supremum is taken only over the cubes $Q(x,r)$, $r > 0$, where $Q(x,r)$ denotes the cube centred at $x$ with side length $r$, we write $M^c_w$. If $w \equiv 1$ we drop the subscript and write $M$ and $M^c$.
\end{definition}

It is well known that $M$ is of weak type $(1,1)$ and strong type $(p,p)$ on $L^p(\mathbb{R}^d)$ for $1 < p \le \infty$ \cite{GrafakosClassical}.
In the light of our $L^p_w(\mathbb{R}^d)$ definition a natural question arises, for fixed $1 < p < \infty$, for which weights $w$ is the sublinear operator $M$ bounded on $L^p_w(\mathbb{R}^d)$? In other words, for which $w$ is there a constant $C$ such that for all $f \in L^p_w(\mathbb{R}^d)$
\begin{equation}\label{eq:question}
  \|Mf\|_{L^p_w} = \left( \int_{\mathbb{R}^d} (Mf)^p\, w\,dx \right)^{1/p} \le C \left( \int_{\mathbb{R}^d} |f|^p\, w\,dx \right)^{1/p} = C\, \|f\|_{L^p_w}.
\end{equation}

\begin{remark}[The zero weight]
If $w = 0$ almost everywhere, then $\|f\|_{L^p_w} = 0$ for every $f$, so $L^p_w(\mathbb{R}^d)$ is trivial and $M$ is trivially bounded.
\end{remark}

\begin{remark}[The endpoint $p = \infty$]
For $p = \infty$ the weight plays essentially no role. If $w > 0$ almost everywhere, then $w(E) = 0$ if and only if $|E| = 0$, so $L^\infty_w(\mathbb{R}^d) = L^\infty(\mathbb{R}^d)$ with equal norms, and $M$ is bounded by the unweighted result. If $w$ is not almost everywhere zero but vanishes on a set $Z$ with $|Z| > 0$, then $\|\chi_Z\|_{L^\infty_w} = 0$ while $M\chi_Z > 0$ everywhere, so $M$ is not bounded. This is why the question \eqref{eq:question} is posed only for $1 < p < \infty$.
\end{remark}

Suppose that \eqref{eq:question} holds for a weight $w$. A first observation is that $w$ cannot vanish on a set of positive measure. The indicator function of a bounded part of such a set has norm zero in $L^p_w$, while its maximal function is positive everywhere, which contradicts \eqref{eq:question}. More generally, $w$ cannot be much smaller on part of a cube than on the rest of it. A function concentrated where $w$ is small has a small norm in $L^p_w$, while its maximal function is large on the whole cube, where $w$ is large. Making this precise leads to the following necessary condition.

\begin{proposition}\label{prop:necessity}
Let $1 < p < \infty$ and let $w$ be a weight that is not almost everywhere zero. If \eqref{eq:question} holds with constant $C$, then for every cube $Q$ we have $\int_Q w^{-\frac{1}{p-1}}\,dx < \infty$ and
\begin{equation}\label{eq:Apcube}
  \left( \frac{1}{|Q|} \int_Q w\,dx \right) \left( \frac{1}{|Q|} \int_Q w^{-\frac{1}{p-1}}\,dx \right)^{p-1} \le C^p .
\end{equation}
In particular, $w > 0$ almost everywhere.
\end{proposition}

Thus, if \eqref{eq:question} holds for a weight $w$, then by Proposition~\ref{prop:necessity}, $w$ satisfies \eqref{eq:Apcube} for every cube $Q$, with the same constant $C$.This gives rise to the next definition by Muckenhoupt~\cite{Muckenhoupt1972}.

\begin{definition}[Muckenhoupt $A_p$ weights]
Let $1 < p < \infty$ and let $w$ be a weight. The \emph{$A_p$ characteristic} of $w$ is
\begin{equation}\label{eq:Ap}
  [w]_{A_p} := \sup_{Q} \left( \frac{1}{|Q|} \int_Q w(x)\,dx \right) \left( \frac{1}{|Q|} \int_Q w(x)^{-\frac{1}{p-1}}\,dx \right)^{p-1},
\end{equation}
where the supremum is taken over all cubes $Q \subset \mathbb{R}^d$. We say that $w \in A_p$ if $[w]_{A_p} < \infty$.
\end{definition}

Taking the supremum over all cubes in Proposition~\ref{prop:necessity} gives the following.

\begin{corollary}\label{cor:necessity}
Let $1 < p < \infty$ and let $w$ be a weight that is not almost everywhere zero. If \eqref{eq:question} holds with constant $C$, then $w \in A_p$ and
\begin{equation}\label{eq:lower}
  [w]_{A_p} \le C^p .
\end{equation}
\end{corollary}

Corollary~\ref{cor:necessity} shows that $w \in A_p$ is a necessary condition for \eqref{eq:question} to hold. It is natural to ask whether it is also sufficient, that is, whether for every $w \in A_p$ there is a constant $C$ such that \eqref{eq:question} holds. Since Corollary~\ref{cor:necessity} moreover shows that any such $C$ must satisfy $C \ge [w]_{A_p}^{1/p}$, we may ask more precisely how large $C$ has to be in terms of $[w]_{A_p}$. The following theorem, due to Buckley \cite{Buckley1993}, answers both questions.

\begin{theorem}[Buckley; Theorem~C in~\cite{Lerner2008}]\label{thm:main}
Let $1 < p < \infty$. There is a constant $c_{p,d}$, depending only on $p$ and $d$, such that for every $w \in A_p$ and every $f \in L^p_w(\mathbb{R}^d)$,
\[
  \|Mf\|_{L^p_w} \le c_{p,d}\, [w]_{A_p}^{\frac{1}{p-1}}\, \|f\|_{L^p_w} .
\]
\end{theorem}

The exponent in Theorem~\ref{thm:main} cannot be improved.

\begin{proposition}[Buckley \cite{Buckley1993}]\label{prop:sharp}
Let $1 < p < \infty$ and $\alpha < \frac{1}{p-1}$. Then there is no constant $c$ such that $\|Mf\|_{L^p_w} \le c\, [w]_{A_p}^{\alpha}\, \|f\|_{L^p_w}$ for all $w \in A_p$ and all $f \in L^p_w(\mathbb{R}^d)$.
\end{proposition}

In particular, Theorem~\ref{thm:main} shows that $w \in A_p$ is also a sufficient condition, since if $[w]_{A_p} < \infty$, then \eqref{eq:question} holds with $C = c_{p,d}[w]_{A_p}^{1/(p-1)}$. Together with Corollary~\ref{cor:necessity}, this answers the question posed at the beginning completely and implies the following result.

\begin{theorem}[Muckenhoupt]\label{thm:muckenhoupt}
Let $1 < p < \infty$ and let $w$ be a weight that is not almost everywhere zero. Then $M$ is bounded on $L^p_w(\mathbb{R}^d)$ if and only if $w \in A_p$.
\end{theorem}

Theorem~\ref{thm:muckenhoupt} is the original result of~\cite{Muckenhoupt1972}, published in 1972. His proof, as well as Buckley's proof of Theorem~\ref{thm:main}, relies on the fact that every $A_p$ weight belongs to $A_{p-\varepsilon}$ for some $\varepsilon > 0$, which is a consequence of the reverse H\"older inequality for $A_p$ weights. Buckley's proof moreover uses interpolation. We present neither of these proofs. For the upper bound in Theorem~\ref{thm:main} we instead present the elementary proof of Lerner \cite{Lerner2008}, which avoids the $A_{p-\varepsilon}$ property, and as noted above Theorem~\ref{thm:muckenhoupt} then needs no separate proof since it follows immidiatly from Corollary~\ref{cor:necessity} and Theorem~\ref{thm:main}. We complement Lerner's argument with proofs of Propositions~\ref{prop:necessity} and~\ref{prop:sharp}.

\section{Proofs}

\subsection{Preliminaries}

\begin{lemma}\label{lem:cubes}
Let $Q$ be a cube with side length $\ell(Q)$ and let $y \in Q$. Then $Q \subseteq Q(y, 2\ell(Q)) \subseteq 3Q$, where $3Q$ denotes the cube with the same centre as $Q$ and side length $3\ell(Q)$.
\end{lemma}

\begin{proof}
Every $z \in Q$ satisfies $|z_i - y_i| \le \ell(Q)$ for $i = 1, \dots, d$, since both $z$ and $y$ lie in $Q$. Hence $Q \subseteq Q(y, 2\ell(Q))$. Let $c$ be the centre of $Q$. Then $|y_i - c_i| \le \frac{1}{2}\ell(Q)$ for $i = 1, \dots, d$, so every $z \in Q(y, 2\ell(Q))$ satisfies
\[
  |z_i - c_i| \le |z_i - y_i| + |y_i - c_i| \le \ell(Q) + \tfrac{1}{2}\ell(Q) = \tfrac{3}{2}\ell(Q)
\]
for $i = 1, \dots, d$, that is, $z \in 3Q$.
\end{proof}

\begin{lemma}\label{lem:centred}
Let $f \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$. Then $Mf(x) \le 2^d M^c f(x)$ for every $x \in \mathbb{R}^d$.
\end{lemma}

\begin{proof}
Let $x \in \mathbb{R}^d$. For a cube $Q$ containing $x$, Lemma~\ref{lem:cubes} gives $Q \subseteq Q(x, 2\ell(Q))$, and since $|Q(x, 2\ell(Q))| = 2^d |Q|$ we obtain
\[
  Mf(x) = \sup_{Q \ni x} \frac{1}{|Q|} \int_Q |f(y)|\,dy \le \sup_{Q \ni x} \frac{2^d}{|Q(x, 2\ell(Q))|} \int_{Q(x, 2\ell(Q))} |f(y)|\,dy \le 2^d M^c f(x),
\]
where the last inequality holds because each $Q(x, 2\ell(Q))$ is one of the cubes in the definition of $M^c f(x)$.
\end{proof}

\begin{lemma}\label{lem:sigma}
Let $1 < p < \infty$ and let $w \in A_p$ be a weight that is not almost everywhere zero. Then $w > 0$ almost everywhere, and $\sigma := w^{-\frac{1}{p-1}}$ is a weight. Moreover $w(Q) > 0$ and $0 < \sigma(Q) < \infty$ for every cube $Q$.
\end{lemma}

\begin{proof}
Suppose that $w = 0$ on a set $E$ with $|E| > 0$. Since $w$ is not almost everywhere zero, there is also a set $F$ with $|F| > 0$ on which $w > 0$. Choose bounded subsets $E' \subseteq E$ and $F' \subseteq F$ of positive measure, and a cube $Q$ containing both. Then $w(Q) \ge w(F') > 0$, while $\sigma = \infty$ on $E'$, so $\sigma(Q) = \infty$. Hence the quantity in \eqref{eq:Ap} is infinite for this cube, which contradicts $w \in A_p$. Thus $w > 0$ almost everywhere, and in particular $w(Q) > 0$ for every cube $Q$. Since $w$ is locally integrable, $w < \infty$ almost everywhere, so $\sigma > 0$ almost everywhere and $\sigma(Q) > 0$ for every cube $Q$. Finally, by the definition of $[w]_{A_p}$,
\[
  \left( \frac{\sigma(Q)}{|Q|} \right)^{p-1} \le [w]_{A_p} \frac{|Q|}{w(Q)} < \infty
\]
for every cube $Q$. Since every compact set is contained in a cube, $\sigma$ is locally integrable.
\end{proof}

\begin{lemma}[{Besicovitch covering lemma \cite[Lemma A.5]{Duoandikoetxea2013}}]\label{lem:besicovitch}
There is a constant $N_d$, depending only on $d$, with the following property. Let $E \subset \mathbb{R}^d$ be a bounded set and, for each $x \in E$, let $r_x > 0$, such that $\sup_{x \in E} r_x < \infty$. Then there are points $x_1, x_2, \ldots \in E$ such that the cubes $Q(x_j, r_{x_j})$ cover $E$ and satisfy $\sum_j \chi_{Q(x_j, r_{x_j})} \le N_d$.
\end{lemma}

\begin{lemma}\label{lem:weak}
Let $v$ be a weight with $v(Q) > 0$ for every cube $Q$, and let $f$ be a measurable function such that $|f|\,v$ is locally integrable. Then $M^c_v f$ is measurable, and
\[
  v\big(\{x \in \mathbb{R}^d : M^c_v f(x) > \lambda\}\big) \le \frac{N_d}{\lambda}\, \|f\|_{L^1_v}
\]
for every $\lambda > 0$, where $N_d$ is the constant in Lemma~\ref{lem:besicovitch}.
\end{lemma}

\begin{proof}
Fix $\lambda > 0$, and call a cube $Q(x,r)$ \emph{good} if
\begin{equation}\label{eq:good}
  \int_{Q(x,r)} |f(y)|\, v(y)\,dy > \lambda\, v(Q(x,r)).
\end{equation}
By the definition of $M^c_v$, we have $M^c_v f(x) > \lambda$ if and only if $Q(x,r)$ is good for some $r > 0$. For fixed $r > 0$, both sides of \eqref{eq:good} are continuous functions of $x$. Indeed, if $x' \to x$, then $\chi_{Q(x',r)} \to \chi_{Q(x,r)}$ outside the boundary of $Q(x,r)$, which has measure zero, and $\chi_{Q(x',r)} \le \chi_{Q(x,r+2)}$ whenever $|x' - x| \le 1$, so the claim follows by dominated convergence since $|f|\,v$ and $v$ are integrable on $Q(x,r+2)$. Hence the set $U_r$ of all $x$ for which $Q(x,r)$ is good is open, and so is
\[
  E_k := \{x : |x| < k\} \cap \bigcup_{0 < r \le k} U_r
\]
for every positive integer $k$. We have $E_1 \subseteq E_2 \subseteq \cdots$, every point of $E_k$ has a good cube, and if $Q(x,r)$ is good, then $x \in E_k$ for every integer $k$ with $k > |x|$ and $k \ge r$. Hence $\{M^c_v f > \lambda\} = \bigcup_k E_k$. In particular this set is open, and since $\lambda > 0$ was arbitrary and $M^c_v f \ge 0$, the function $M^c_v f$ is measurable.

For each $x \in E_k$ we choose $r_x \in (0,k]$ such that $Q(x,r_x)$ is good. Since $E_k$ is bounded and $r_x \le k$ for every $x \in E_k$, Lemma~\ref{lem:besicovitch} gives points $x_1, x_2, \ldots \in E_k$ such that the cubes $Q(x_j, r_{x_j})$ cover $E_k$ and satisfy $\sum_j \chi_{Q(x_j, r_{x_j})} \le N_d$. Using \eqref{eq:good} for each $Q(x_j, r_{x_j})$ and monotone convergence to interchange sum and integral, we obtain
\[
\begin{aligned}
  v(E_k) &\le \sum_j v\big(Q(x_j, r_{x_j})\big) \le \frac{1}{\lambda} \sum_j \int_{Q(x_j, r_{x_j})} |f|\, v\,dy \\
  &= \frac{1}{\lambda} \int_{\mathbb{R}^d} \Big( \sum_j \chi_{Q(x_j, r_{x_j})} \Big) |f|\, v\,dy \le \frac{N_d}{\lambda}\, \|f\|_{L^1_v}.
\end{aligned}
\]
Since $E_1 \subseteq E_2 \subseteq \cdots$ and $\bigcup_k E_k = \{M^c_v f > \lambda\}$, continuity of the measure $v(x)\,dx$ from below gives the claim.
\end{proof}

\begin{lemma}\label{lem:key}
Let $v$ be a weight with $v(Q) > 0$ for every cube $Q$, and let $1 < q < \infty$. Then there is a constant $C_{q,d}$, depending only on $q$ and $d$, such that
\[
  \|M^c_v f\|_{L^q_v} \le C_{q,d}\, \|f\|_{L^q_v}
\]
for all $f \in L^q_v(\mathbb{R}^d)$.
\end{lemma}

\begin{proof}
If $f = f_1 + f_2$ with $f_1 \in L^1_v(\mathbb{R}^d)$ and $f_2 \in L^\infty_v(\mathbb{R}^d)$, then $\int_Q |f|\, v\,dy \le \|f_1\|_{L^1_v} + \|f_2\|_{L^\infty_v}\, v(Q) < \infty$ for every cube $Q$. Hence $M^c_v$ is defined on $L^1_v(\mathbb{R}^d) + L^\infty_v(\mathbb{R}^d)$, and $M^c_v f$ is measurable by Lemma~\ref{lem:weak}. Since $|f + h| \le |f| + |h|$, the operator $M^c_v$ is sublinear.

By Lemma~\ref{lem:weak}, $v(\{M^c_v f > \lambda\}) \le \frac{N_d}{\lambda} \|f\|_{L^1_v}$ for every $f \in L^1_v(\mathbb{R}^d)$ and $\lambda > 0$. Moreover, if $f \in L^\infty_v(\mathbb{R}^d)$, then $|f| \le \|f\|_{L^\infty_v}$ outside a set of $v$-measure zero, so every average in the definition of $M^c_v f$ is at most $\|f\|_{L^\infty_v}$, and $\|M^c_v f\|_{L^\infty_v} \le \|f\|_{L^\infty_v}$. The Marcinkiewicz interpolation theorem \cite[Theorem A.6]{Duoandikoetxea2013}, applied on the measure space $(\mathbb{R}^d, v(x)\,dx)$ with exponents $1$ and $\infty$, now gives the claim with a constant depending only on $q$, $N_d$ and $1$, and hence only on $q$ and $d$.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:main}}

We prove Theorem~\ref{thm:main} following the argument of Lerner \cite{Lerner2008}.

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Let $1 < p < \infty$, let $w \in A_p$ and let $f \in L^p_w(\mathbb{R}^d)$. If $w = 0$ almost everywhere, both sides of the inequality vanish, so we may assume that $w$ is not almost everywhere zero. By Lemma~\ref{lem:sigma}, $w > 0$ almost everywhere and $\sigma := w^{-\frac{1}{p-1}}$ is a weight with $w(Q) > 0$ and $0 < \sigma(Q) < \infty$ for every cube $Q$. In particular, the operators $M^c_w$ and $M^c_\sigma$ are defined.

We first prove a pointwise estimate for $Mf$. Fix $x \in \mathbb{R}^d$ and $r > 0$, let $Q := Q(x,r)$, and let $3Q$ be as in Lemma~\ref{lem:cubes}. Multiplying and dividing by $w(Q)\,\sigma(3Q)^{p-1}$, we obtain
\begin{equation}\label{eq:identity}
\begin{aligned}
  \left( \frac{1}{|Q|} \int_Q |f(y)|\,dy \right)^{p-1}
  &= \frac{1}{|Q|^{p-1}} \left( \int_Q |f(y)|\,dy \right)^{p-1} \\
  &= \frac{w(Q)\,\sigma(3Q)^{p-1}}{|Q|^p} \cdot \frac{|Q|}{w(Q)} \cdot \frac{1}{\sigma(3Q)^{p-1}} \left( \int_Q |f(y)|\,dy \right)^{p-1} \\
  &= \frac{w(Q)\,\sigma(3Q)^{p-1}}{|Q|^p} \cdot \frac{|Q|}{w(Q)} \left( \frac{1}{\sigma(3Q)} \int_Q |f(y)|\,dy \right)^{p-1}.
\end{aligned}
\end{equation}
Since $w(Q) \le w(3Q)$ and $|3Q| = 3^d |Q|$, the first factor on the right-hand side satisfies
\[
  \frac{w(Q)\,\sigma(3Q)^{p-1}}{|Q|^p} \le 3^{dp} \left( \frac{1}{|3Q|} \int_{3Q} w(y)\,dy \right) \left( \frac{1}{|3Q|} \int_{3Q} \sigma(y)\,dy \right)^{p-1} \le 3^{dp}\, [w]_{A_p}.
\]
For the second factor, let $y \in Q$. By Lemma~\ref{lem:cubes}, $Q \subseteq Q(y, 2\ell(Q)) \subseteq 3Q$, so
\[
\begin{aligned}
  \frac{1}{\sigma(3Q)} \int_Q |f(z)|\,dz
  &= \frac{1}{\sigma(3Q)} \int_Q |f(z)|\,\sigma(z)^{-1}\,\sigma(z)\,dz \\
  &\le \frac{1}{\sigma(Q(y, 2\ell(Q)))} \int_{Q(y, 2\ell(Q))} |f(z)|\,\sigma(z)^{-1}\,\sigma(z)\,dz \le M^c_\sigma\big(|f|\sigma^{-1}\big)(y).
\end{aligned}
\]
This holds for every $y \in Q$, and therefore
\[
\begin{aligned}
  \frac{|Q|}{w(Q)} \left( \frac{1}{\sigma(3Q)} \int_Q |f(z)|\,dz \right)^{p-1}
  &= \frac{1}{w(Q)} \int_Q \left( \frac{1}{\sigma(3Q)} \int_Q |f(z)|\,dz \right)^{p-1} dy \\
  &\le \frac{1}{w(Q)} \int_Q \Big( M^c_\sigma\big(|f|\sigma^{-1}\big)(y) \Big)^{p-1} w(y)^{-1}\,w(y)\,dy \\
  &\le M^c_w\Big( \big(M^c_\sigma(|f|\sigma^{-1})\big)^{p-1} w^{-1} \Big)(x),
\end{aligned}
\]
where the last inequality holds because $Q = Q(x,r)$. Inserting both bounds into \eqref{eq:identity}, taking the supremum over $r > 0$ and applying Lemma~\ref{lem:centred}, we arrive at
\begin{equation}\label{eq:pointwise}
  Mf(x) \le 2^d\, 3^{\frac{dp}{p-1}}\, [w]_{A_p}^{\frac{1}{p-1}} \Big( M^c_w\Big( \big(M^c_\sigma(|f|\sigma^{-1})\big)^{p-1} w^{-1} \Big)(x) \Big)^{\frac{1}{p-1}} .
\end{equation}

We now take norms. Since $1 - \frac{p}{p-1} = -\frac{1}{p-1}$, we have $w^{1 - \frac{p}{p-1}} = \sigma$, and similarly $\sigma^{1-p} = w$. The second identity gives
\[
  \big\| |f|\sigma^{-1} \big\|_{L^p_\sigma}^p = \int_{\mathbb{R}^d} |f(x)|^p\, \sigma(x)^{-p}\, \sigma(x)\,dx = \int_{\mathbb{R}^d} |f(x)|^p\, w(x)\,dx = \|f\|_{L^p_w}^p ,
\]
so $|f|\sigma^{-1} \in L^p_\sigma(\mathbb{R}^d)$. Using the first identity and Lemma~\ref{lem:key} with $v = \sigma$ and $q = p$, we obtain
\[
\begin{aligned}
  \int_{\mathbb{R}^d} \Big( \big(M^c_\sigma(|f|\sigma^{-1})(x)\big)^{p-1} w(x)^{-1} \Big)^{\frac{p}{p-1}} w(x)\,dx
  &= \int_{\mathbb{R}^d} \big(M^c_\sigma(|f|\sigma^{-1})(x)\big)^{p}\, \sigma(x)\,dx \\
  &\le C_{p,d}^p\, \|f\|_{L^p_w}^p ,
\end{aligned}
\]
so the function $\big(M^c_\sigma(|f|\sigma^{-1})\big)^{p-1} w^{-1}$ lies in $L^{p/(p-1)}_w(\mathbb{R}^d)$. Finally, we raise \eqref{eq:pointwise} to the power $p$, integrate against $w$, and apply Lemma~\ref{lem:key} with $v = w$ and $q = \frac{p}{p-1}$. Together with the previous estimate this gives
\[
  \|Mf\|_{L^p_w}^p \le \Big( 2^d\, 3^{\frac{dp}{p-1}} \Big)^p\, [w]_{A_p}^{\frac{p}{p-1}}\, C_{\frac{p}{p-1},d}^{\frac{p}{p-1}}\, C_{p,d}^p\, \|f\|_{L^p_w}^p .
\]
Taking $p$-th roots, we conclude that
\[
  \|Mf\|_{L^p_w} \le c_{p,d}\, [w]_{A_p}^{\frac{1}{p-1}}\, \|f\|_{L^p_w}, \qquad c_{p,d} := 2^d\, 3^{\frac{dp}{p-1}}\, C_{\frac{p}{p-1},d}^{\frac{1}{p-1}}\, C_{p,d},
\]
and $c_{p,d}$ depends only on $p$ and $d$.
\end{proof}

\subsection{Sharpness of the exponent}

\begin{proof}[Proof of Proposition~\ref{prop:sharp}]
Let $1 < p < \infty$ and $\alpha < \frac{1}{p-1}$, and suppose, for contradiction, that there is a constant $c$ such that
\begin{equation}\label{eq:contra}
  \|Mf\|_{L^p_w} \le c\, [w]_{A_p}^{\alpha}\, \|f\|_{L^p_w}
\end{equation}
for all $w \in A_p$ and all $f \in L^p_w(\mathbb{R}^d)$. For every weight $w$ that is not almost everywhere zero and every cube $Q$, H\"older's inequality with exponents $p$ and $\frac{p}{p-1}$ gives
\[
  1 = \frac{1}{|Q|} \int_Q w^{\frac{1}{p}}\, w^{-\frac{1}{p}}\,dy \le \left( \frac{1}{|Q|} \int_Q w\,dy \right)^{\frac{1}{p}} \left( \frac{1}{|Q|} \int_Q w^{-\frac{1}{p-1}}\,dy \right)^{\frac{p-1}{p}},
\]
so $[w]_{A_p} \ge 1$. Hence \eqref{eq:contra} remains true if $\alpha$ is replaced by $\max\{\alpha, 0\}$, and we may assume that $0 \le \alpha < \frac{1}{p-1}$.

For $0 < \delta < 1$ we consider the weight $w_\delta(x) := |x|^{(d-\delta)(p-1)}$, for which $w_\delta^{-\frac{1}{p-1}}(x) = |x|^{\delta-d}$. By polar coordinates,
\begin{equation}\label{eq:polar}
  \int_{\{|y| < r\}} |y|^{\delta - d}\,dy = \omega_d\, \frac{r^\delta}{\delta} \qquad \text{for every } r > 0,
\end{equation}
where $\omega_d$ denotes the surface measure of the unit sphere in $\mathbb{R}^d$.

We first show that $w_\delta \in A_p$ with
\begin{equation}\label{eq:wdelta}
  [w_\delta]_{A_p} \le C\, \delta^{1-p},
\end{equation}
where $C$ depends only on $p$ and $d$. The function $w_\delta$ is continuous and positive outside the origin, so it is a weight that is not almost everywhere zero. Write $a := (d-\delta)(p-1) \ge 0$, and let $Q := Q(c,\ell)$ be a cube. Every $y \in Q$ satisfies $|y - c| \le \frac{\sqrt{d}}{2}\,\ell$.

Suppose first that $|c| \le \sqrt{d}\,\ell$. Then every $y \in Q$ satisfies $|y| \le R\ell$ with $R := \frac{3\sqrt{d}}{2}$. Since $a \ge 0$, this gives $w_\delta \le (R\ell)^a$ on $Q$, and by \eqref{eq:polar},
\[
  \frac{1}{|Q|} \int_Q |y|^{\delta-d}\,dy \le \frac{1}{\ell^d} \int_{\{|y| < R\ell\}} |y|^{\delta-d}\,dy = \frac{\omega_d\, R^\delta\, \ell^{\delta-d}}{\delta}.
\]
Since $a + (\delta - d)(p-1) = 0$, we obtain
\[
  \left( \frac{1}{|Q|} \int_Q w_\delta\,dy \right) \left( \frac{1}{|Q|} \int_Q |y|^{\delta-d}\,dy \right)^{p-1} \le (R\ell)^a \left( \frac{\omega_d\, R^\delta\, \ell^{\delta-d}}{\delta} \right)^{p-1} = R^{d(p-1)}\, \omega_d^{\,p-1}\, \delta^{1-p}.
\]

Suppose instead that $|c| > \sqrt{d}\,\ell$. Then $|y - c| < \frac{1}{2}|c|$, so $\frac{1}{2}|c| \le |y| \le \frac{3}{2}|c|$ for every $y \in Q$. Hence $w_\delta \le \big(\tfrac{3}{2}|c|\big)^a$ and $|y|^{\delta-d} \le \big(\tfrac{1}{2}|c|\big)^{\delta-d}$ on $Q$, and
\[
  \left( \frac{1}{|Q|} \int_Q w_\delta\,dy \right) \left( \frac{1}{|Q|} \int_Q |y|^{\delta-d}\,dy \right)^{p-1} \le \Big( \tfrac{3}{2} \Big)^a\, 2^{(d-\delta)(p-1)} = 3^a \le 3^{d(p-1)} \le 3^{d(p-1)}\, \delta^{1-p}.
\]
Taking the supremum over all cubes proves \eqref{eq:wdelta} with $C := \max\{R^{d(p-1)}\omega_d^{\,p-1},\, 3^{d(p-1)}\}$.

Next we construct a function for which $M$ is large in $L^p_{w_\delta}$. Let $f_\delta(x) := |x|^{\delta-d}\chi_{Q(0,2)}(x)$. Then $f_\delta \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$, and since $Q(0,2) \subseteq \{|y| < 2\sqrt{d}\}$, \eqref{eq:polar} gives
\[
  \|f_\delta\|_{L^p_{w_\delta}}^p = \int_{Q(0,2)} |y|^{(\delta-d)p}\, |y|^{(d-\delta)(p-1)}\,dy = \int_{Q(0,2)} |y|^{\delta-d}\,dy \le \omega_d\, \frac{(2\sqrt{d})^\delta}{\delta} < \infty .
\]
Moreover $\|f_\delta\|_{L^p_{w_\delta}} > 0$, so $f_\delta \in L^p_{w_\delta}(\mathbb{R}^d)$ is not the zero element. Let $x \in Q(0,2)$ with $x \neq 0$, and put $r := \max_{1 \le i \le d} |x_i|$, so that $0 < r \le |x|$. The cube $Q(0,2r)$ contains $x$, is contained in $Q(0,2)$, and contains the ball $\{|y| < r\}$. By the definition of $M$ and \eqref{eq:polar},
\[
  Mf_\delta(x) \ge \frac{1}{(2r)^d} \int_{Q(0,2r)} |y|^{\delta-d}\,dy \ge \frac{1}{(2r)^d} \int_{\{|y| < r\}} |y|^{\delta-d}\,dy = \frac{\omega_d}{2^d\delta}\, r^{\delta-d} \ge \frac{\omega_d}{2^d\delta}\, |x|^{\delta-d},
\]
where the last inequality holds because $\delta - d < 0$ and $r \le |x|$. Thus $Mf_\delta \ge \frac{\omega_d}{2^d\delta} f_\delta$ almost everywhere on $Q(0,2)$, and this also holds outside $Q(0,2)$, where $f_\delta = 0$. Consequently
\begin{equation}\label{eq:Mfdelta}
  \|Mf_\delta\|_{L^p_{w_\delta}} \ge \frac{\omega_d}{2^d\delta}\, \|f_\delta\|_{L^p_{w_\delta}} .
\end{equation}

Finally we combine these estimates. Applying \eqref{eq:contra} to $w_\delta$ and $f_\delta$, and using \eqref{eq:Mfdelta}, \eqref{eq:wdelta} and $\alpha \ge 0$, we obtain
\[
  \frac{\omega_d}{2^d\delta}\, \|f_\delta\|_{L^p_{w_\delta}} \le \|Mf_\delta\|_{L^p_{w_\delta}} \le c\, [w_\delta]_{A_p}^{\alpha}\, \|f_\delta\|_{L^p_{w_\delta}} \le c\, C^\alpha\, \delta^{\alpha(1-p)}\, \|f_\delta\|_{L^p_{w_\delta}} .
\]
Dividing by $\|f_\delta\|_{L^p_{w_\delta}} \in (0,\infty)$ and multiplying by $\delta$ gives
\[
  \frac{\omega_d}{2^d} \le c\, C^\alpha\, \delta^{1 - \alpha(p-1)} \qquad \text{for every } 0 < \delta < 1 .
\]
Since $\alpha < \frac{1}{p-1}$, the exponent $1 - \alpha(p-1)$ is positive, so the right-hand side tends to $0$ as $\delta \to 0$, while the left-hand side is a positive constant independent of $\delta$. This is a contradiction, and the proof is complete.
\end{proof}
