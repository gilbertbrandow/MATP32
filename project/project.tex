\begin{abstract}
In 1972, B.~Muckenhoupt \cite{Muckenhoupt1972} proved that the Hardy--Littlewood maximal operator is bounded on the weighted space $L^p_w(\mathbb{R}^d)$, $1 < p < \infty$, if and only if $w$ satisfies the $A_p$ condition. In 1993, S.~Buckley \cite{Buckley1993} showed that its operator norm is bounded by a constant times $[w]_{A_p}^{1/(p-1)}$, and that this exponent is best possible. In 2008, A.~K.~Lerner \cite{Lerner2008} gave an alternative proof of Buckley's bound. We present Lerner's proof in detail, together with Buckley's example showing that the exponent cannot be improved.
\end{abstract}

\section{Introduction}

\begin{definition}[Weights]
A \emph{weight} is a function $w \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$ with $w \geq 0$. For a measurable set $E \subseteq \mathbb{R}^d$ we write $w(E) := \int_E w(x)\,dx$. Then $E \mapsto w(E)$ is a measure on the Lebesgue measurable sets.
\end{definition}

\begin{definition}[Weighted $L^p$ spaces]\label{def:Lp}
Let $w$ be a weight and let $1 \le p < \infty$. We define
\[
  L^p_w(\mathbb{R}^d) := \left\{ f : \mathbb{R}^d \to \mathbb{R} \text{ measurable} : \int_{\mathbb{R}^d} |f(x)|^p\, w(x)\,dx < \infty \right\},
\]
with norm $\|f\|_{L^p_w} := \left( \int_{\mathbb{R}^d} |f(x)|^p\, w(x)\,dx \right)^{1/p}$. Moreover, $L^\infty_w(\mathbb{R}^d)$ is the space of measurable functions $f$ with $\|f\|_{L^\infty_w} := \inf\{ t \ge 0 : w(\{|f| > t\}) = 0 \} < \infty$.
\end{definition}

The central object of this report is the Hardy--Littlewood maximal operator $M$. Our proofs also use its analogue with respect to the measure $w(x)\,dx$.

\begin{definition}[Hardy--Littlewood maximal operator]\label{def:maximal}
Let $w$ be a weight with $w(Q) > 0$ for every cube $Q$. For a measurable function $f$ and $x \in \mathbb{R}^d$,
\[
  M_w f(x) := \sup_{Q \ni x} \frac{1}{w(Q)} \int_Q |f(y)|\, w(y)\,dy,
\]
where the supremum is taken over all closed cubes $Q \subset \mathbb{R}^d$ with sides parallel to the axes containing $x$. All cubes in this report are of this kind. If the supremum is taken only over the cubes $Q(x,r) := \{y : |y - x|_\infty \le r/2\}$, $r > 0$, where $|y|_\infty := \max_{1 \le k \le d} |y_k|$, we write $M^c_w$. If $w \equiv 1$ we drop the subscript and write $M$ and $M^c$.
\end{definition}

It is well known that $M$ is of weak type $(1,1)$ and strong type $(p,p)$ for $1 < p \le \infty$ \cite[Theorem~1.1]{Duoandikoetxea2013}. For $1 < p < \infty$ and a weight $w$ we say that $M$ is bounded on $L^p_w(\mathbb{R}^d)$ if there is a constant $C$ such that
\begin{equation}\label{eq:question}
  \|Mf\|_{L^p_w} = \left( \int_{\mathbb{R}^d} (Mf)^p\, w\,dx \right)^{1/p} \le C \left( \int_{\mathbb{R}^d} |f|^p\, w\,dx \right)^{1/p} = C\, \|f\|_{L^p_w}
\end{equation}
for all $f \in L^p_w(\mathbb{R}^d)$.

\begin{remark}[The zero weight]\label{rem:zero}
If $w = 0$ almost everywhere, then $\|f\|_{L^p_w} = 0$ for every $f$, so $L^p_w(\mathbb{R}^d)$ is trivial and $M$ is trivially bounded.
\end{remark}

\begin{remark}[The endpoint $p = \infty$]
For $p = \infty$, the operator $M$ is bounded on $L^\infty_w(\mathbb{R}^d)$ if and only if $w > 0$ almost everywhere or $w = 0$ almost everywhere. Muckenhoupt \cite[Theorem~3]{Muckenhoupt1972} proves this on an interval, and the same argument works in $\mathbb{R}^d$. This is why we only consider $1 < p < \infty$.
\end{remark}

Suppose that \eqref{eq:question} holds and let $Q$ be a cube. Since $Mf(x)$ is at least the average of $|f|$ over $Q$ for $x \in Q$, writing out the norms in \eqref{eq:question} by Definition~\ref{def:Lp} gives
\[
  \left( \frac{1}{|Q|} \int_Q |f|\,dx \right)^p w(Q) \le \int_Q (Mf)^p\, w\,dx \le C^p \int_{\mathbb{R}^d} |f|^p\, w\,dx
\]
for every $f \in L^p_w(\mathbb{R}^d)$. Each choice of $f$ thus gives a condition on $w$, and the choice $f = w^{-\frac{1}{p-1}}\chi_Q$ makes this condition as strong as possible. The following proposition makes this precise, and we prove it in Appendix~\ref{app:necessity}.

\begin{proposition}\label{prop:necessity}
Let $1 < p < \infty$ and let $w$ be a weight that is not almost everywhere zero. If \eqref{eq:question} holds with constant $C$, then for every cube $Q$ we have $\int_Q w^{-\frac{1}{p-1}}\,dx < \infty$ and
\begin{equation}\label{eq:Apcube}
  \left( \frac{1}{|Q|} \int_Q w\,dx \right) \left( \frac{1}{|Q|} \int_Q w^{-\frac{1}{p-1}}\,dx \right)^{p-1} \le C^p .
\end{equation}
In particular, $w > 0$ almost everywhere.
\end{proposition}

This motivates the following definition \cite[(7.10)]{Muckenhoupt1972}.

\begin{definition}[Muckenhoupt $A_p$ weights]
Let $1 < p < \infty$ and let $w$ be a weight. The \emph{$A_p$ characteristic} of $w$ is
\begin{equation}\label{eq:Ap}
  [w]_{A_p} := \sup_{Q} \left( \frac{1}{|Q|} \int_Q w(x)\,dx \right) \left( \frac{1}{|Q|} \int_Q w(x)^{-\frac{1}{p-1}}\,dx \right)^{p-1},
\end{equation}
where the supremum is taken over all cubes $Q \subset \mathbb{R}^d$. We say that $w \in A_p$ if $[w]_{A_p} < \infty$.
\end{definition}

\begin{definition}[Dual weight]\label{def:dual}
Let $w$ be a weight and let $1 < p < \infty$. The \emph{dual weight} of $w$ is $\sigma := w^{-\frac{1}{p-1}}$, with $\sigma = \infty$ where $w = 0$ \cite[Section~1]{Buckley1993}. For measurable sets $E$ we write $\sigma(E) := \int_E \sigma\,dx$.
\end{definition}

With this notation \eqref{eq:Ap} takes the more compact form
\[
  [w]_{A_p} = \sup_Q \frac{w(Q)}{|Q|} \left( \frac{\sigma(Q)}{|Q|} \right)^{p-1}.
\]

Taking the supremum over all cubes in Proposition~\ref{prop:necessity} gives the following.

\begin{corollary}\label{cor:necessity}
Let $1 < p < \infty$ and let $w$ be a weight that is not almost everywhere zero. If \eqref{eq:question} holds with constant $C$, then $w \in A_p$ and
\begin{equation}\label{eq:lower}
  [w]_{A_p} \le C^p .
\end{equation}
\end{corollary}

Muckenhoupt proved that the condition $w \in A_p$ is also sufficient.

\begin{theoremA}[{Muckenhoupt \cite[Theorem~9]{Muckenhoupt1972}}]\label{thm:muckenhoupt}
Let $1 < p < \infty$ and let $w$ be a weight that is not almost everywhere zero. Then $M$ is bounded on $L^p_w(\mathbb{R}^d)$ if and only if $w \in A_p$.
\end{theoremA}

Theorem~\ref{thm:muckenhoupt} shows that the operator norm of $M$ on $L^p_w(\mathbb{R}^d)$, that is, the smallest constant $C$ in \eqref{eq:question}, is finite precisely when $w \in A_p$. By Corollary~\ref{cor:necessity}, this norm is bounded from below by $[w]_{A_p}^{1/p}$. The natural question is therefore whether it is also bounded from above by a power of $[w]_{A_p}$, and which power is the smallest possible. This was answered by Buckley \cite[Theorem~2.5]{Buckley1993}.

\begin{theorem}[Buckley]\label{thm:main}
Let $1 < p < \infty$. There is a constant $c_{p,d}$, depending only on $p$ and $d$, such that for every $w \in A_p$ and every $f \in L^p_w(\mathbb{R}^d)$,
\[
  \|Mf\|_{L^p_w} \le c_{p,d}\, [w]_{A_p}^{\frac{1}{p-1}}\, \|f\|_{L^p_w} .
\]
\end{theorem}

\begin{proposition}[Buckley]\label{prop:sharp}
Let $1 < p < \infty$ and $\alpha < \frac{1}{p-1}$. Then there is no constant $c$ such that $\|Mf\|_{L^p_w} \le c\, [w]_{A_p}^{\alpha}\, \|f\|_{L^p_w}$ for all $w \in A_p$ and all $f \in L^p_w(\mathbb{R}^d)$.
\end{proposition}

Buckley's proof of Theorem~\ref{thm:main} \cite[proof of Theorem~2.5]{Buckley1993}, like Muckenhoupt's proof of Theorem~\ref{thm:muckenhoupt} \cite[Lemma~5 and Theorem~2]{Muckenhoupt1972}, relies on a deep property that every $A_p$ weight belongs to $A_{p-\varepsilon}$ for some $\varepsilon > 0$ \cite[Theorem~4.5]{Duoandikoetxea2013}, together with interpolation. Lerner \cite[Theorem~C]{Lerner2008} found a short and elementary proof of Theorem~\ref{thm:main}. We present his argument in detail and complete it with a proof of Proposition~\ref{prop:sharp}, based on Buckley's example in $\mathbb{R}$ \cite[proof of Theorem~2.5]{Buckley1993}, which we extend to $\mathbb{R}^d$.

\begin{remark}[An alternative proof of Theorem~\ref{thm:muckenhoupt}]
If $M$ is bounded on $L^p_w(\mathbb{R}^d)$, then $w \in A_p$ by Corollary~\ref{cor:necessity}, which follows from Proposition~\ref{prop:necessity}. Conversely, if $w \in A_p$, then $M$ is bounded on $L^p_w(\mathbb{R}^d)$ by Theorem~\ref{thm:main}. Combined, these results imply Theorem~\ref{thm:muckenhoupt}.
\end{remark}

\section{Proofs}\label{sec:proofs}

\subsection{Preliminaries}

\begin{lemma}\label{lem:cubes}
Let $Q$ be a cube with side length $\ell(Q)$ and let $y \in Q$. Then $Q \subseteq Q(y, 2\ell(Q)) \subseteq 3Q$, where $3Q$ denotes the cube with the same centre as $Q$ and side length $3\ell(Q)$.
\end{lemma}

\begin{proof}
Every $z \in Q$ satisfies $|z - y|_\infty \le \ell(Q)$, since both $z$ and $y$ lie in $Q$. Hence $Q \subseteq Q(y, 2\ell(Q))$. If $c$ is the centre of $Q$, then $|y - c|_\infty \le \frac{1}{2}\ell(Q)$, so every $z \in Q(y, 2\ell(Q))$ satisfies $|z - c|_\infty \le |z - y|_\infty + |y - c|_\infty \le \frac{3}{2}\ell(Q)$, that is, $z \in 3Q$.
\end{proof}

\begin{lemma}\label{lem:centred}
Let $f \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$. Then $Mf(x) \le 2^d M^c f(x)$ for every $x \in \mathbb{R}^d$.
\end{lemma}

\begin{proof}
Let $x \in \mathbb{R}^d$. For a cube $Q$ containing $x$, Lemma~\ref{lem:cubes} gives $Q \subseteq Q(x, 2\ell(Q))$, and since $|Q(x, 2\ell(Q))| = 2^d |Q|$ we obtain
\begin{align*}
  Mf(x) &= \sup_{Q \ni x} \frac{1}{|Q|} \int_Q |f(y)|\,dy \\
  &\le \sup_{Q \ni x} \frac{2^d}{|Q(x, 2\ell(Q))|} \int_{Q(x, 2\ell(Q))} |f(y)|\,dy \\
  &\le 2^d \sup_{r > 0} \frac{1}{|Q(x, r)|} \int_{Q(x, r)} |f(y)|\,dy \\
  &= 2^d M^c f(x).
\end{align*}
\end{proof}

\begin{lemma}\label{lem:sigma}
Let $1 < p < \infty$ and let $w \in A_p$ be a weight that is not almost everywhere zero. Then $w > 0$ almost everywhere and $\sigma$, as in Definition~\ref{def:dual}, is a weight. Moreover $w(Q) > 0$ and $0 < \sigma(Q) < \infty$ for every cube $Q$, and $L^p_w(\mathbb{R}^d) \subseteq L^1_{\mathrm{loc}}(\mathbb{R}^d)$.
\end{lemma}

\begin{proof}
Let $Q$ be a cube with $w(Q) > 0$. By the definition of $[w]_{A_p}$,
\begin{equation}\label{eq:sigmafinite}
  \left( \frac{\sigma(Q)}{|Q|} \right)^{p-1} \le [w]_{A_p} \frac{|Q|}{w(Q)} < \infty,
\end{equation}
so $\sigma(Q) < \infty$ on any cube $Q$ where $w(Q) > 0$.

Assume, for contradiction, that $w = 0$ on a set $E$ with $|E| > 0$. Since $w$ is not almost everywhere zero, $w > 0$ on a set $F$ with $|F| > 0$, so we can choose a cube $Q$ with $|Q \cap E| > 0$ and $|Q \cap F| > 0$. Then $w(Q) \ge w(Q \cap F) > 0$, so $\sigma(Q) < \infty$ by \eqref{eq:sigmafinite}. But $\sigma = \infty$ on $Q \cap E$ by Definition~\ref{def:dual}, so $\sigma(Q) = \infty$, a contradiction. Therefore $w > 0$ almost everywhere. In particular $w(Q) > 0$, and hence $\sigma(Q) < \infty$ by \eqref{eq:sigmafinite}, for every cube $Q$.

Since $w(Q) < \infty$ for every cube $Q$, we have $w < \infty$ almost everywhere, so $\sigma > 0$ almost everywhere and $\sigma(Q) > 0$ for every cube $Q$. As every compact set lies in a cube, $\sigma$ is locally integrable and hence a weight. Finally, for $f \in L^p_w(\mathbb{R}^d)$ and every cube $Q$, applying H\"older's inequality gives
\[
  \int_Q |f|\,dx = \int_Q |f|\, w^{\frac{1}{p}}\, w^{-\frac{1}{p}}\,dx \le \Big( \int_Q |f|^p\, w\,dx \Big)^{\frac{1}{p}} \Big( \int_Q w^{-\frac{1}{p-1}}\,dx \Big)^{\frac{p-1}{p}} \le \|f\|_{L^p_w}\, \sigma(Q)^{\frac{p-1}{p}},
\]
which is finite.
\end{proof}

\begin{remark}[The Vitali covering lemma]
In the proof of the weak type $(1,1)$ estimate for $M$ \cite[Theorem~1.1]{Duoandikoetxea2013}, a compact set $K \subseteq \{Mf > t\}$ is covered by finitely many cubes $Q$ with $\frac{1}{|Q|}\int_Q |f|\,dx > t$. The Vitali covering lemma \cite[Lemma~A.4]{Duoandikoetxea2013}, which holds for balls as well as cubes, selects disjoint cubes $Q_1, \ldots, Q_N$ among them with $K \subseteq \bigcup_j 3Q_j$. Since $|3Q_j| = 3^d |Q_j|$, this gives
\[
  |K| \le \sum_{j=1}^N |3Q_j| = 3^d \sum_{j=1}^N |Q_j| \le \frac{3^d}{t} \int_{\mathbb{R}^d} |f|\,dx.
\]
For the measure $v(x)\,dx$, the equality $|3Q_j| = 3^d|Q_j|$ would have to be replaced by an inequality $v(3Q_j) \le C\,v(Q_j)$, and no constant $C$ works for all weights $v$. For example, if $v = \varepsilon$ on $Q_j$ and $v = 1$ on the rest of $3Q_j$, then $v(3Q_j)/v(Q_j) \ge (3^d - 1)/\varepsilon$, which is arbitrarily large. The Besicovitch covering lemma covers $K$ by the selected cubes themselves, without enlarging them, and therefore gives a constant independent of $v$.
\end{remark}

\begin{lemma}[{Besicovitch covering lemma \cite[Theorem~1.1]{Guzman1975}}]\label{lem:besicovitch}
Let $E \subset \mathbb{R}^d$ be a bounded set, and suppose that for each $x \in E$ a cube $Q(x, r_x)$ centred at $x$ is given. Then there are points $x_1, x_2, \ldots \in E$ such that the cubes $Q(x_j, r_{x_j})$ cover $E$ and every point of $\mathbb{R}^d$ lies in at most $N_d$ of them, where $N_d$ depends only on $d$.
\end{lemma}

\begin{proof}
Assume that $E$ is nonempty and let $D := \sup_{x,y \in E} |x - y|_\infty < \infty$. If $r_x \ge 2D$ for some $x \in E$, then $E \subseteq Q(x, r_x)$ and this single cube suffices. Hence assume that $r_x < 2D$ for every $x \in E$.

Let $s_0 := \sup_{x \in E} r_x$ and choose $x_1 \in E$ with $r_{x_1} > s_0/2$. Given $x_1, \ldots, x_m$, let
\[
  s_m := \sup\Big\{ r_x : x \in E \setminus \bigcup_{j \le m} Q(x_j, r_{x_j}) \Big\}
\]
and choose $x_{m+1} \in E \setminus \bigcup_{j \le m} Q(x_j, r_{x_j})$ with $r_{x_{m+1}} > s_m/2$. The selection stops if this set is empty. If $j < i$, then $x_i$ lies outside $Q(x_j, r_{x_j})$, and $x_i$ was a possible choice when $x_j$ was chosen. Hence
\begin{equation}\label{eq:besA}
  |x_i - x_j|_\infty > \frac{r_{x_j}}{2} \qquad \text{and} \qquad r_{x_i} \le s_{j-1} < 2 r_{x_j} .
\end{equation}

If the selection stops, the cubes $Q(x_j, r_{x_j})$ cover $E$ by construction. Assume that it does not stop, and suppose, for contradiction, that some $x \in E$ is not covered. Then $x$ is a possible choice at every step, so $r_x \le s_{j-1} < 2r_{x_j}$ for every $j$. If $j < i$, then by \eqref{eq:besA}
\[
  \frac{r_{x_i}}{8} + \frac{r_{x_j}}{8} < \frac{3 r_{x_j}}{8} < |x_i - x_j|_\infty ,
\]
so the cubes $Q(x_j, r_{x_j}/4)$ are pairwise disjoint. Since $x_j \in E$ and $r_{x_j} < 2D$, they are all contained in $Q(x_1, 4D)$. Comparing volumes, for every $N$,
\[
  N \left( \frac{r_x}{8} \right)^d < \sum_{j=1}^N \left( \frac{r_{x_j}}{4} \right)^d \le (4D)^d ,
\]
which is false for large $N$. Hence every $x \in E$ is covered.

Fix $z \in \mathbb{R}^d$. For $\varepsilon \in \{-1, 1\}^d$ let
\[
  O_\varepsilon := \{ y \in \mathbb{R}^d : \varepsilon_k (y_k - z_k) \ge 0 \text{ for } k = 1, \ldots, d \}.
\]
Every $y \in \mathbb{R}^d$ lies in $O_\varepsilon$ with $\varepsilon_k := 1$ if $y_k \ge z_k$ and $\varepsilon_k := -1$ otherwise, so these $2^d$ orthants cover $\mathbb{R}^d$. If $y, y' \in O_\varepsilon$, then $y_k - z_k$ and $y'_k - z_k$ have the same sign for every $k$, and hence
\begin{equation}\label{eq:orthant}
  |y - y'|_\infty \le \max\{ |y - z|_\infty,\ |y' - z|_\infty \} .
\end{equation}

For each $\varepsilon$ let
\[
  J_\varepsilon := \{ x_i : x_i \in O_\varepsilon \text{ and } z \in Q(x_i, r_{x_i}) \}.
\]
Every cube $Q(x_i, r_{x_i})$ containing $z$ has its centre in some $O_\varepsilon$, so the number of such cubes is at most the total number of points in the sets $J_\varepsilon$.

Fix $\varepsilon$ with $J_\varepsilon$ nonempty and let $j := \min\{ i : x_i \in J_\varepsilon \}$, so that $x_j$ is the point of $J_\varepsilon$ that was chosen first. Let $n$ be the number of points in $J_\varepsilon \setminus \{x_j\}$, so that $J_\varepsilon$ has $1 + n$ points. Let $x_i \in J_\varepsilon \setminus \{x_j\}$, so that $i > j$. Since $z$ lies in both $Q(x_i, r_{x_i})$ and $Q(x_j, r_{x_j})$, \eqref{eq:besA} and \eqref{eq:orthant} give
\[
  \frac{r_{x_j}}{2} < |x_i - x_j|_\infty \le \max\{ |x_i - z|_\infty,\ |x_j - z|_\infty \} \le \max\Big\{ \frac{r_{x_i}}{2},\ \frac{r_{x_j}}{2} \Big\}.
\]
Hence $r_{x_i} > r_{x_j}$, and by \eqref{eq:besA} also $|x_i - z|_\infty \le r_{x_i}/2 < r_{x_j}$.

For each $x_i \in J_\varepsilon \setminus \{x_j\}$ consider the cube $Q(x_i, r_{x_j}/2)$. If $x_i, x_m \in J_\varepsilon \setminus \{x_j\}$ with $m < i$, then $r_{x_m} > r_{x_j}$, so \eqref{eq:besA} gives $|x_i - x_m|_\infty > r_{x_m}/2 > r_{x_j}/2$. Thus the centres of these $n$ cubes are more than their side length apart, and the cubes are pairwise disjoint. Moreover, for $x_i \in J_\varepsilon \setminus \{x_j\}$ every $y \in Q(x_i, r_{x_j}/2)$ satisfies
\[
  |y - z|_\infty \le |y - x_i|_\infty + |x_i - z|_\infty < \frac{r_{x_j}}{4} + r_{x_j} = \frac{5 r_{x_j}}{4},
\]
so all of them are contained in $Q(z, 5r_{x_j}/2)$. Since they are disjoint, their total volume is at most the volume of $Q(z, 5r_{x_j}/2)$, that is,
\[
  n \left( \frac{r_{x_j}}{2} \right)^d \le \left( \frac{5 r_{x_j}}{2} \right)^d = 5^d \left( \frac{r_{x_j}}{2} \right)^d .
\]
Hence $n \le 5^d$ and $J_\varepsilon$ has at most $1 + 5^d$ points. Summing over the $2^d$ orthants, $z$ lies in at most $2^d(1 + 5^d)$ of the cubes $Q(x_i, r_{x_i})$, and the lemma holds with $N_d := 2^d(1 + 5^d)$.
\end{proof}

\begin{lemma}\label{lem:weak}
Let $v$ be a weight with $v(Q) > 0$ for every cube $Q$, and let $f$ be a measurable function with $|f|\,v \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$. Then $M^c_v f$ is measurable, and for every $t > 0$,
\[
  v\big(\{x \in \mathbb{R}^d : M^c_v f(x) > t\}\big) \le \frac{N_d}{t}\, \|f\|_{L^1_v},
\]
where $N_d$ is the constant in Lemma~\ref{lem:besicovitch}.
\end{lemma}

\begin{proof}
Let $r > 0$ and $x \in \mathbb{R}^d$, and let $(x_n)$ be a sequence converging to $x$. If $|y - x|_\infty < r/2$, then $|y - x_n|_\infty < r/2$ for all large $n$. If $|y - x|_\infty > r/2$, then $|y - x_n|_\infty > r/2$ for all large $n$. Hence $\chi_{Q(x_n, r)}(y) \to \chi_{Q(x, r)}(y)$ for every $y$ outside the boundary of $Q(x, r)$, which is a set of measure zero. Moreover, $R := \sup_n |x_n - x|_\infty$ is finite, and $Q(x_n, r) \subseteq Q(x, r + 2R)$ for every $n$, so $\chi_{Q(x_n, r)}\,|f|\,v \le \chi_{Q(x, r+2R)}\,|f|\,v$, which is integrable since $|f|\,v \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$. By dominated convergence,
\[
  \lim_{n\to\infty} \int_{Q(x_n, r)} |f|\, v\,dy = \lim_{n\to\infty} \int_{\mathbb{R}^d} \chi_{Q(x_n, r)}\,|f|\, v\,dy = \int_{\mathbb{R}^d} \chi_{Q(x, r)}\,|f|\, v\,dy = \int_{Q(x, r)} |f|\, v\,dy .
\]
Thus $x \mapsto \int_{Q(x,r)} |f|\,v\,dy$ is sequentially continuous, and hence continuous. Taking $f \equiv 1$ shows that $x \mapsto v(Q(x,r))$ is continuous as well. Since $v(Q(x,r)) > 0$, the average
\[
  x \mapsto \frac{1}{v(Q(x,r))} \int_{Q(x,r)} |f|\, v\,dy
\]
is continuous for every $r > 0$. For $t > 0$ the set $\{M^c_v f > t\}$ is the union over $r > 0$ of the sets where this average exceeds $t$. These are open, so $\{M^c_v f > t\}$ is open. Since $t > 0$ was arbitrary and $M^c_v f \ge 0$, the function $M^c_v f$ is measurable.

Now fix $t > 0$. For $k = 1, 2, \ldots$ let $E_k := \{x : |x| < k \text{ and } M^c_v f(x) > t\}$. Let $x \in E_k$. Since $M^c_v f(x) > t$, the definition of $M^c_v$ gives some $r > 0$ with $\frac{1}{v(Q(x,r))}\int_{Q(x,r)} |f|\,v\,dy > t$. Choose such an $r$ and call it $r_x$. Since $E_k$ is bounded, Lemma~\ref{lem:besicovitch} gives points $x_1, x_2, \ldots \in E_k$ such that the cubes $Q(x_j, r_{x_j})$ cover $E_k$ and $\sum_j \chi_{Q(x_j, r_{x_j})} \le N_d$. Hence
\begin{align*}
  v(E_k) &\le \sum_j v\big(Q(x_j, r_{x_j})\big) \le \frac{1}{t} \sum_j \int_{Q(x_j, r_{x_j})} |f|\, v\,dy \\
  &= \frac{1}{t} \int_{\mathbb{R}^d} \Big( \sum_j \chi_{Q(x_j, r_{x_j})} \Big) |f|\, v\,dy \le \frac{N_d}{t}\, \|f\|_{L^1_v},
\end{align*}
where the second inequality holds by the choice of $r_{x_j}$, and the equality follows from monotone convergence. Since $E_1 \subseteq E_2 \subseteq \cdots$ and $v$ is a measure, the bound for $v(E_k)$ gives
\[
  v\big(\{M^c_v f > t\}\big) = v\Big(\bigcup_k E_k\Big) = \lim_{k\to\infty} v(E_k) \le \frac{N_d}{t}\,\|f\|_{L^1_v}.
\]
\end{proof}

\begin{lemma}\label{lem:key}
Let $v$ be a weight with $v(Q) > 0$ for every cube $Q$, and let $1 < q < \infty$. Then there is a constant $C_{q,d}$, depending only on $q$ and $d$, such that
\[
  \|M^c_v f\|_{L^q_v} \le C_{q,d}\, \|f\|_{L^q_v}
\]
for all $f \in L^q_v(\mathbb{R}^d)$.
\end{lemma}

\begin{proof}
Let $f = f_1 + f_2$ with $f_1 \in L^1_v(\mathbb{R}^d)$ and $f_2 \in L^\infty_v(\mathbb{R}^d)$. By the definition of $\|f_2\|_{L^\infty_v}$, the sets $\{|f_2| > \|f_2\|_{L^\infty_v} + 1/n\}$, $n = 1, 2, \ldots$, have $v$-measure zero, and hence so does their union $\{|f_2| > \|f_2\|_{L^\infty_v}\}$. This implies
\begin{equation}\label{eq:Linfty}
  \int_Q |f_2|\,v\,dy \le \|f_2\|_{L^\infty_v}\, v(Q) \quad \text{for every cube } Q,
\end{equation}
and therefore
\[
  \int_Q |f|\,v\,dy \le \int_Q |f_1|\,v\,dy + \int_Q |f_2|\,v\,dy \le \|f_1\|_{L^1_v} + \|f_2\|_{L^\infty_v}\, v(Q) < \infty .
\]
Hence $|f|\,v \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$, so $M^c_v f$ is defined and, by Lemma~\ref{lem:weak}, measurable.

Let $f, h \in L^1_v(\mathbb{R}^d) + L^\infty_v(\mathbb{R}^d)$, $x \in \mathbb{R}^d$ and $r > 0$. The pointwise inequality $|f + h| \le |f| + |h|$ gives
\begin{align*}
  \frac{1}{v(Q(x,r))}\int_{Q(x,r)} |f + h|\,v\,dy &\le \frac{1}{v(Q(x,r))}\int_{Q(x,r)} |f|\,v\,dy + \frac{1}{v(Q(x,r))}\int_{Q(x,r)} |h|\,v\,dy \\
  &\le M^c_v f(x) + M^c_v h(x).
\end{align*}
The right-hand side does not depend on $r$, so taking the supremum over $r > 0$ yields $M^c_v(f + h)(x) \le M^c_v f(x) + M^c_v h(x)$. For a scalar $\lambda$,
\begin{align*}
  M^c_v(\lambda f)(x) &= \sup_{r > 0} \frac{1}{v(Q(x,r))}\int_{Q(x,r)} |\lambda|\,|f|\,v\,dy \\
  &= |\lambda| \sup_{r > 0} \frac{1}{v(Q(x,r))}\int_{Q(x,r)} |f|\,v\,dy = |\lambda|\, M^c_v f(x),
\end{align*}
where the second equality holds since $|\lambda| \ge 0$. Hence $M^c_v$ is sublinear.

By Lemma~\ref{lem:weak}, $v(\{M^c_v f > t\}) \le \frac{N_d}{t} \|f\|_{L^1_v}$ for every $f \in L^1_v(\mathbb{R}^d)$ and $t > 0$. For $f \in L^\infty_v(\mathbb{R}^d)$, \eqref{eq:Linfty} with $f_2 = f$ shows that every average in the definition of $M^c_v f(x)$ is at most $\|f\|_{L^\infty_v}$. Hence $M^c_v f(x) \le \|f\|_{L^\infty_v}$ for every $x \in \mathbb{R}^d$, and in particular $\|M^c_v f\|_{L^\infty_v} \le \|f\|_{L^\infty_v}$.

The Marcinkiewicz interpolation theorem \cite[Theorem A.6]{Duoandikoetxea2013}, applied on the measure space $(\mathbb{R}^d, v(x)\,dx)$ with exponents $1$ and $\infty$, now gives the claim with a constant depending only on $q$, $N_d$ and $1$, and hence only on $q$ and $d$.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:main}}
\begin{proof}[Proof of Theorem~\ref{thm:main}]
Let $1 < p < \infty$, let $w \in A_p$ and let $f \in L^p_w(\mathbb{R}^d)$. For $w = 0$ almost everywhere, $\|Mf\|_{L^p_w} = \|f\|_{L^p_w} = 0$ by Remark~\ref{rem:zero}, and the inequality holds trivially. Assume therefore that $w$ is not almost everywhere zero. Then, by Lemma~\ref{lem:sigma}, $w > 0$ almost everywhere and $\sigma$ is a weight with $w(Q) > 0$ and $0 < \sigma(Q) < \infty$ for every cube $Q$. Hence $M^c_w$ and $M^c_\sigma$ are defined, and by Lemma~\ref{lem:key} they are of strong type $(q,q)$ on $L^q_w(\mathbb{R}^d)$ and $L^q_\sigma(\mathbb{R}^d)$, respectively, for every $1 < q < \infty$.

For $x \in \mathbb{R}^d$, $r > 0$ and $y \in Q(x,r)$, Lemma~\ref{lem:cubes} gives $Q(x,r) \subseteq Q(y, 2r) \subseteq Q(x, 3r)$, so
\begin{equation}\label{eq:inner}
  \frac{1}{\sigma(Q(x,3r))} \int_{Q(x,r)} |f|\,dz \le \frac{1}{\sigma(Q(y,2r))} \int_{Q(y,2r)} |f|\,\sigma^{-1}\,\sigma\,dz \le M^c_\sigma\big(|f|\sigma^{-1}\big)(y).
\end{equation}
The inequality $w(Q(x,r)) \le w(Q(x,3r))$, the equality $|Q(x,3r)| = 3^d\,|Q(x,r)|$ and the definition of $[w]_{A_p}$ imply
\begin{equation}\label{eq:Apfactor}
  \frac{w(Q(x,r))\,\sigma(Q(x,3r))^{p-1}}{|Q(x,r)|^p} \le 3^{dp}\, \frac{w(Q(x,3r))\,\sigma(Q(x,3r))^{p-1}}{|Q(x,3r)|^p} \le 3^{dp}\, [w]_{A_p}.
\end{equation}
Multiplying and dividing by $w(Q(x,r))\,\sigma(Q(x,3r))^{p-1}$, and then bounding the two marked factors by \eqref{eq:Apfactor} and \eqref{eq:inner}, gives
\begin{align*}
  \MoveEqLeft \left( \frac{1}{|Q(x,r)|} \int_{Q(x,r)} |f|\,dz \right)^{p-1} \\
  &= \underbrace{\frac{w(Q(x,r))\,\sigma(Q(x,3r))^{p-1}}{|Q(x,r)|^p}}_{\le\, 3^{dp} [w]_{A_p} \text{ by } \eqref{eq:Apfactor}} \cdot \frac{1}{w(Q(x,r))} \int_{Q(x,r)} \Bigg( \underbrace{\frac{1}{\sigma(Q(x,3r))} \int_{Q(x,r)} |f|\,dz}_{\le\, M^c_\sigma(|f|\sigma^{-1})(y) \text{ by } \eqref{eq:inner}} \Bigg)^{p-1} dy \\
  &\le 3^{dp}\, [w]_{A_p} \cdot \frac{1}{w(Q(x,r))} \int_{Q(x,r)} \Big( M^c_\sigma\big(|f|\sigma^{-1}\big)(y) \Big)^{p-1} w(y)^{-1}\, w(y)\,dy \\
  &\le 3^{dp}\, [w]_{A_p}\, M^c_w\Big( \big(M^c_\sigma(|f|\sigma^{-1})\big)^{p-1} w^{-1} \Big)(x).
\end{align*}
By the definition of $M^c f$, taking the supremum over $r > 0$ gives
\begin{equation}\label{eq:pointwise}
\begin{aligned}
  M^c f(x) &= \sup_{r > 0} \frac{1}{|Q(x,r)|} \int_{Q(x,r)} |f|\,dz \\
  &\le 3^{\frac{dp}{p-1}}\, [w]_{A_p}^{\frac{1}{p-1}} \Big( M^c_w\Big( \big(M^c_\sigma(|f|\sigma^{-1})\big)^{p-1} w^{-1} \Big)(x) \Big)^{\frac{1}{p-1}} .
\end{aligned}
\end{equation}

The identities $w^{1 - \frac{p}{p-1}} = \sigma$ and $\sigma^{1-p} = w$, together with Lemma~\ref{lem:key} with $v = \sigma$ and $q = p$, give
\begin{equation}\label{eq:innernorm}
  \Big\| \big(M^c_\sigma(|f|\sigma^{-1})\big)^{p-1} w^{-1} \Big\|_{L^{p/(p-1)}_w}^{\frac{1}{p-1}} = \big\| M^c_\sigma\big(|f|\sigma^{-1}\big) \big\|_{L^p_\sigma} \le C_{p,d}\, \big\| |f|\sigma^{-1} \big\|_{L^p_\sigma} = C_{p,d}\, \|f\|_{L^p_w} .
\end{equation}
In particular $\big(M^c_\sigma(|f|\sigma^{-1})\big)^{p-1} w^{-1} \in L^{p/(p-1)}_w(\mathbb{R}^d)$. Lemma~\ref{lem:sigma} gives $f \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$, so Lemma~\ref{lem:centred} gives the first inequality below, and bounding the marked factors as indicated gives
\begin{align*}
  \|Mf\|_{L^p_w}
  &\le 2^d\, \Big\| \underbrace{M^c f}_{\mathclap{\text{bounded by } \eqref{eq:pointwise}}} \Big\|_{L^p_w} \\
  &\le 2^d\, \Big\| 3^{\frac{dp}{p-1}}\, [w]_{A_p}^{\frac{1}{p-1}} \Big( M^c_w\Big( \big(M^c_\sigma(|f|\sigma^{-1})\big)^{p-1} w^{-1} \Big) \Big)^{\frac{1}{p-1}} \Big\|_{L^p_w} \\
  &= 2^d\, 3^{\frac{dp}{p-1}}\, [w]_{A_p}^{\frac{1}{p-1}}\, \underbrace{\Big\| M^c_w\Big( \big(M^c_\sigma(|f|\sigma^{-1})\big)^{p-1} w^{-1} \Big) \Big\|_{L^{p/(p-1)}_w}^{\frac{1}{p-1}}}_{\text{bounded by Lemma~\ref{lem:key} with } v = w,\ q = \frac{p}{p-1}} \\
  &\le 2^d\, 3^{\frac{dp}{p-1}}\, C_{\frac{p}{p-1},d}^{\frac{1}{p-1}}\, [w]_{A_p}^{\frac{1}{p-1}}\, \underbrace{\Big\| \big(M^c_\sigma(|f|\sigma^{-1})\big)^{p-1} w^{-1} \Big\|_{L^{p/(p-1)}_w}^{\frac{1}{p-1}}}_{\text{bounded by } \eqref{eq:innernorm}} \\
  &\le 2^d\, 3^{\frac{dp}{p-1}}\, C_{\frac{p}{p-1},d}^{\frac{1}{p-1}}\, [w]_{A_p}^{\frac{1}{p-1}}\, C_{p,d}\, \|f\|_{L^p_w}.
\end{align*}
\end{proof}

\subsection{Sharpness of the exponent}

\begin{proof}[Proof of Proposition~\ref{prop:sharp}]
Let $1 < p < \infty$ and $\alpha < \frac{1}{p-1}$, and suppose, for contradiction, that there is a constant $c$ such that
\begin{equation}\label{eq:contra}
  \|Mf\|_{L^p_w} \le c\, [w]_{A_p}^{\alpha}\, \|f\|_{L^p_w}
\end{equation}
for all $w \in A_p$ and all $f \in L^p_w(\mathbb{R}^d)$. For every weight $w$ that is not almost everywhere zero and every cube $Q$, H\"older's inequality with exponents $p$ and $\frac{p}{p-1}$ gives
\[
  1 = \frac{1}{|Q|} \int_Q w^{\frac{1}{p}}\, w^{-\frac{1}{p}}\,dy \le \left( \frac{1}{|Q|} \int_Q w\,dy \right)^{\frac{1}{p}} \left( \frac{1}{|Q|} \int_Q w^{-\frac{1}{p-1}}\,dy \right)^{\frac{p-1}{p}},
\]
so $[w]_{A_p} \ge 1$. Hence \eqref{eq:contra} remains true if $\alpha$ is replaced by $\max\{\alpha, 0\}$, and we may assume that $0 \le \alpha < \frac{1}{p-1}$.

For $0 < \delta < 1$ we consider the weight $w_\delta(x) := |x|^{(d-\delta)(p-1)}$, for which $w_\delta^{-\frac{1}{p-1}}(x) = |x|^{\delta-d}$. By polar coordinates,
\begin{equation}\label{eq:polar}
  \int_{\{|y| < r\}} |y|^{\delta - d}\,dy = \omega_d\, \frac{r^\delta}{\delta} \qquad \text{for every } r > 0,
\end{equation}
where $\omega_d$ denotes the surface measure of the unit sphere in $\mathbb{R}^d$.

We first show that $w_\delta \in A_p$ with
\begin{equation}\label{eq:wdelta}
  [w_\delta]_{A_p} \le C\, \delta^{1-p},
\end{equation}
where $C$ depends only on $p$ and $d$. The function $w_\delta$ is continuous and positive outside the origin, so it is a weight that is not almost everywhere zero. Write $a := (d-\delta)(p-1) \ge 0$, and let $Q := Q(c,\ell)$ be a cube. Every $y \in Q$ satisfies $|y - c| \le \frac{\sqrt{d}}{2}\,\ell$.

Suppose first that $|c| \le \sqrt{d}\,\ell$. Then every $y \in Q$ satisfies $|y| \le R\ell$ with $R := \frac{3\sqrt{d}}{2}$. Since $a \ge 0$, this gives $w_\delta \le (R\ell)^a$ on $Q$, and by \eqref{eq:polar},
\[
  \frac{1}{|Q|} \int_Q |y|^{\delta-d}\,dy \le \frac{1}{\ell^d} \int_{\{|y| < R\ell\}} |y|^{\delta-d}\,dy = \frac{\omega_d\, R^\delta\, \ell^{\delta-d}}{\delta}.
\]
Since $a + (\delta - d)(p-1) = 0$, we obtain
\[
  \left( \frac{1}{|Q|} \int_Q w_\delta\,dy \right) \left( \frac{1}{|Q|} \int_Q |y|^{\delta-d}\,dy \right)^{p-1} \le (R\ell)^a \left( \frac{\omega_d\, R^\delta\, \ell^{\delta-d}}{\delta} \right)^{p-1} = R^{d(p-1)}\, \omega_d^{\,p-1}\, \delta^{1-p}.
\]

Suppose instead that $|c| > \sqrt{d}\,\ell$. Then $|y - c| < \frac{1}{2}|c|$, so $\frac{1}{2}|c| \le |y| \le \frac{3}{2}|c|$ for every $y \in Q$. Hence $w_\delta \le \big(\tfrac{3}{2}|c|\big)^a$ and $|y|^{\delta-d} \le \big(\tfrac{1}{2}|c|\big)^{\delta-d}$ on $Q$, and
\[
  \left( \frac{1}{|Q|} \int_Q w_\delta\,dy \right) \left( \frac{1}{|Q|} \int_Q |y|^{\delta-d}\,dy \right)^{p-1} \le \Big( \tfrac{3}{2} \Big)^a\, 2^{(d-\delta)(p-1)} = 3^a \le 3^{d(p-1)} \le 3^{d(p-1)}\, \delta^{1-p}.
\]
Taking the supremum over all cubes proves \eqref{eq:wdelta} with $C := \max\{R^{d(p-1)}\omega_d^{\,p-1},\, 3^{d(p-1)}\}$.

Next we construct a function for which $M$ is large in $L^p_{w_\delta}$. Let $f_\delta(x) := |x|^{\delta-d}\chi_{Q(0,2)}(x)$. Then $f_\delta \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$, and since $Q(0,2) \subseteq \{|y| < 2\sqrt{d}\}$, \eqref{eq:polar} gives
\[
  \|f_\delta\|_{L^p_{w_\delta}}^p = \int_{Q(0,2)} |y|^{(\delta-d)p}\, |y|^{(d-\delta)(p-1)}\,dy = \int_{Q(0,2)} |y|^{\delta-d}\,dy \le \omega_d\, \frac{(2\sqrt{d})^\delta}{\delta} < \infty .
\]
Moreover $\|f_\delta\|_{L^p_{w_\delta}} > 0$, so $f_\delta \in L^p_{w_\delta}(\mathbb{R}^d)$ is not the zero element. Let $x \in Q(0,2)$ with $x \neq 0$, and put $r := |x|_\infty$, so that $0 < r \le |x|$. The cube $Q(0,2r)$ contains $x$, is contained in $Q(0,2)$, and contains the ball $\{|y| < r\}$. By the definition of $M$ and \eqref{eq:polar},
\[
  Mf_\delta(x) \ge \frac{1}{(2r)^d} \int_{Q(0,2r)} |y|^{\delta-d}\,dy \ge \frac{1}{(2r)^d} \int_{\{|y| < r\}} |y|^{\delta-d}\,dy = \frac{\omega_d}{2^d\delta}\, r^{\delta-d} \ge \frac{\omega_d}{2^d\delta}\, |x|^{\delta-d},
\]
where the last inequality holds because $\delta - d < 0$ and $r \le |x|$. Thus $Mf_\delta \ge \frac{\omega_d}{2^d\delta} f_\delta$ almost everywhere on $Q(0,2)$, and this also holds outside $Q(0,2)$, where $f_\delta = 0$. Consequently
\begin{equation}\label{eq:Mfdelta}
  \|Mf_\delta\|_{L^p_{w_\delta}} \ge \frac{\omega_d}{2^d\delta}\, \|f_\delta\|_{L^p_{w_\delta}} .
\end{equation}

Finally we combine these estimates. Applying \eqref{eq:contra} to $w_\delta$ and $f_\delta$, and using \eqref{eq:Mfdelta}, \eqref{eq:wdelta} and $\alpha \ge 0$, we obtain
\[
  \frac{\omega_d}{2^d\delta}\, \|f_\delta\|_{L^p_{w_\delta}} \le \|Mf_\delta\|_{L^p_{w_\delta}} \le c\, [w_\delta]_{A_p}^{\alpha}\, \|f_\delta\|_{L^p_{w_\delta}} \le c\, C^\alpha\, \delta^{\alpha(1-p)}\, \|f_\delta\|_{L^p_{w_\delta}} .
\]
Dividing by $\|f_\delta\|_{L^p_{w_\delta}} \in (0,\infty)$ and multiplying by $\delta$ gives
\[
  \frac{\omega_d}{2^d} \le c\, C^\alpha\, \delta^{1 - \alpha(p-1)} \qquad \text{for every } 0 < \delta < 1 .
\]
Since $\alpha < \frac{1}{p-1}$, the exponent $1 - \alpha(p-1)$ is positive, so the right-hand side tends to $0$ as $\delta \to 0$, while the left-hand side is a positive constant independent of $\delta$. This is a contradiction, and the proof is complete.
\end{proof}

\section{Proof of Proposition~\ref{prop:necessity}}\label{app:necessity}

We follow the proof of \cite[Theorem~1.7]{Duoandikoetxea2013}.

\begin{proof}
We first show that $w > 0$ almost everywhere. Fix a cube $Q$ and a measurable set $S \subseteq Q$. For $x \in Q$ the cube $Q$ is one of the cubes in the supremum defining $M\chi_S(x)$, so $M\chi_S(x) \ge |S|/|Q|$. Hence \eqref{eq:question} applied to $f = \chi_S$ gives
\[
  \left( \frac{|S|}{|Q|} \right)^p w(Q) \le C^p\, w(S).
\]
Since $w$ is not almost everywhere zero, every bounded set $S$ is contained in a cube $Q$ with $w(Q) > 0$, so $w(S) = 0$ implies $|S| = 0$. Applying this to the sets $\{w = 0\} \cap \{|x| < k\}$, $k = 1, 2, \ldots$, shows that $w > 0$ almost everywhere. In particular $w(Q) > 0$ for every cube $Q$.

Now fix a cube $Q$ and $\varepsilon > 0$, and let $f := (w + \varepsilon)^{-\frac{1}{p-1}}\, \chi_Q$. Since $w \le w + \varepsilon$,
\[
  (w + \varepsilon)^{-\frac{p}{p-1}}\, w \le (w + \varepsilon)^{-\frac{p}{p-1}}\, (w + \varepsilon) = (w + \varepsilon)^{-\frac{1}{p-1}} \le \varepsilon^{-\frac{1}{p-1}},
\]
so $f \in L^p_w(\mathbb{R}^d)$. Moreover, for $x \in Q$ the cube $Q$ is one of the cubes in the supremum defining $Mf(x)$, so $Mf(x) \ge \frac{1}{|Q|} \int_Q (w + \varepsilon)^{-\frac{1}{p-1}}\,dy$, and therefore
\begin{align*}
  \left( \frac{1}{|Q|} \int_Q (w + \varepsilon)^{-\frac{1}{p-1}}\,dy \right)^p w(Q)
  &= \int_Q \left( \frac{1}{|Q|} \int_Q (w + \varepsilon)^{-\frac{1}{p-1}}\,dy \right)^p w(x)\,dx \\
  &\le \int_Q (Mf)^p\, w\,dx \\
  &\le \|Mf\|_{L^p_w}^p \\
  &\le C^p\, \|f\|_{L^p_w}^p \\
  &= C^p \int_Q (w + \varepsilon)^{-\frac{p}{p-1}}\, w\,dx \\
  &\le C^p \int_Q (w + \varepsilon)^{-\frac{1}{p-1}}\,dx .
\end{align*}
Since $w$ is locally integrable, $w < \infty$ almost everywhere, so $(w + \varepsilon)^{-\frac{1}{p-1}} > 0$ almost everywhere and the integral on the right-hand side is positive. Dividing by it we obtain
\[
  \frac{w(Q)}{|Q|} \left( \frac{1}{|Q|} \int_Q (w + \varepsilon)^{-\frac{1}{p-1}}\,dy \right)^{p-1} \le C^p .
\]
As $\varepsilon \to 0$, $(w + \varepsilon)^{-\frac{1}{p-1}}$ increases to $w^{-\frac{1}{p-1}}$. Hence
\begin{align*}
  \MoveEqLeft \frac{w(Q)}{|Q|} \left( \frac{1}{|Q|} \int_Q w^{-\frac{1}{p-1}}\,dy \right)^{p-1} \\
  &= \frac{w(Q)}{|Q|} \left( \frac{1}{|Q|} \int_Q \lim_{\varepsilon \to 0} (w + \varepsilon)^{-\frac{1}{p-1}}\,dy \right)^{p-1} \\
  &= \lim_{\varepsilon \to 0}\, \frac{w(Q)}{|Q|} \left( \frac{1}{|Q|} \int_Q (w + \varepsilon)^{-\frac{1}{p-1}}\,dy \right)^{p-1} \qquad \text{(monotone convergence)} \\
  &\le C^p ,
\end{align*}
which is \eqref{eq:Apcube}. Since $w(Q) > 0$, this also shows that $\int_Q w^{-\frac{1}{p-1}}\,dy < \infty$.
\end{proof}
