\noindent{\small\textbf{The sharp $A_p$ bound for the Hardy--Littlewood maximal operator} \hfill Simon Gustafsson}

\begin{itemize}
  \item Cubes are closed with sides parallel to the axes. A cube $Q$ has side length $\ell(Q)$, and $3Q$ is the cube with the same centre and side length $3\ell(Q)$.
  \item $Q(x,r) := \{y : |y - x|_\infty \le r/2\}$ is the cube centred at $x$ with side length $r$, where $|z|_\infty := \max_{k} |z_k|$.
  \item $v(E) := \int_E v\,dy$ for measurable $v \ge 0$ and measurable $E \subseteq \mathbb{R}^d$. In particular $|E|$ is the Lebesgue measure of $E$.
\end{itemize}

\begin{definition}[Weights]
A \emph{weight} is a function $w \in L^1_{\mathrm{loc}}(\mathbb{R}^d)$ with $w \ge 0$ that is not almost everywhere zero.
\end{definition}
\begin{definition}[Weighted $L^p$ spaces]
$\|f\|_{L^p_w} := \big( \int_{\mathbb{R}^d} |f|^p\, w\,dy \big)^{1/p}$, and $L^p_w(\mathbb{R}^d)$ consists of the $f$ with $\|f\|_{L^p_w} < \infty$.
\end{definition}
\begin{definition}[Maximal operators]
\[
  M_w f(x) := \sup_{Q \ni x} \frac{1}{w(Q)} \int_Q |f|\, w\,dy, \qquad
  M^c_w f(x) := \sup_{r > 0} \frac{1}{w(Q(x,r))} \int_{Q(x,r)} |f|\, w\,dy .
\]
For $w \equiv 1$ these are $M$ and $M^c$.
\end{definition}
\begin{definition}[$A_p$ characteristic]
\[
  [w]_{A_p} := \sup_{Q} \left( \frac{1}{|Q|} \int_Q w\,dy \right) \left( \frac{1}{|Q|} \int_Q w^{-\frac{1}{p-1}}\,dy \right)^{p-1},
\]
and $w \in A_p$ if $[w]_{A_p} < \infty$.
\end{definition}
\begin{definition}[Dual weight]
$\sigma := w^{-\frac{1}{p-1}}$, so that
$\displaystyle [w]_{A_p} = \sup_Q \frac{w(Q)}{|Q|} \left( \frac{\sigma(Q)}{|Q|} \right)^{p-1}$.
\end{definition}
